## Global Constraints

- **判据在测量之前写死**（`docs/rf/backend_contract.md` §5.3）。事后再定判据等于没有判据。
  凡未实测的项一律标 **NOT RUN**，不写"应该通过"。

- Produces: `axi_ctrl_0` 偏移 **0x10** 的只读 STATUS 字，位域：
  `[31:28]=0`（保留）、`[27:24]=WIN_LOG2=15`、`[23:8]=win_cycles`、
  `[7:4]=win_seq`、`[3]=win_active`、`[2]=rst_source_ok 同步后电平`、
  `[1]=win_done`、`[0]=clk_seen`（粘滞：曾经见到过 DATA_CLK 边沿）。
  `f_DATA_CLK = win_cycles / 2^WIN_LOG2 × f_FCLK0 = win_cycles / 32768 × 50e6 Hz`。
  后续 Task 4/5/6 的 `rf_main.c` 与 Task 8 的审计都读这个字。

**为什么计数器必须由 `data_clk` 时钟驱动，而不是被 FCLK0 采样：**
FCLK0 = 50 MHz，两个候选 `f_DATA_CLK` 是 30 / 60 MHz。采样要求
`f_FCLK0 > 2 × f_DATA_CLK` ⇒ `f_DATA_CLK < 25 MHz`，**两个候选都不满足**。
被采样的话计数器会丢边沿却仍然读出一个"像样"的数（别名），错得看不出来。
格雷码只保证**多比特跨域安全**（任一次采样得到旧值或新值，绝不是乱码），不解决丢边沿。
所以：**计数器在 DATA_CLK 域跑，只把格雷码跨界**。

- [ ] **Step 1: 写失败的 TB**

创建 `rf_overlay/tb/axi_ctrl_tb.v`：

```verilog
`timescale 1 ns / 1 ps
//////////////////////////////////////////////////////////////////////////////////
// axi_ctrl_tb -- self-checking testbench for the read-only STATUS register.
//
// The register answers one question: is the AD9361's DATA_CLK actually reaching
// the FPGA, and how fast?  A DATA_CLK-domain Gray counter is sampled by a
// self-running 2^WIN_LOG2-cycle FCLK0 window; software reads the count together
// with a rolling sequence number so it never reads a torn window.
//
// The decisive test is the NEGATIVE one: with data_clk held still the register
// must report zero cycles and clk_seen=0.  A counter accidentally clocked by
// FCLK0 would pass every positive test and fail this one -- so data_clk is
// deliberately driven at a DIFFERENT rate from aclk (33.333 ns vs 20 ns).
//////////////////////////////////////////////////////////////////////////////////

module axi_ctrl_tb;

	localparam integer AW       = 6;
	localparam real    ACLK_NS  = 20.0;      // FCLK0 = 50 MHz
	localparam real    DCLK_NS  = 33.333;    // DATA_CLK = 30 MHz (candidate A)
	localparam integer WIN_LOG2 = 15;
	localparam real    WIN_NS   = 32768.0 * ACLK_NS;          // 655.36 us
	localparam real    EXP_EDGES = WIN_NS / DCLK_NS;          // ~19660.8

	reg aclk = 1'b0, aresetn = 1'b0;
	reg data_clk = 1'b0;
	reg dclk_en  = 1'b0;
	reg rst_source_ok = 1'b0;

	// AXI-Lite
	reg  [AW-1:0] awaddr = 0;
	reg           awvalid = 0;
	wire          awready;
	reg  [31:0]   wdata = 0;
	reg  [3:0]    wstrb = 4'hF;
	reg           wvalid = 0;
	wire          wready;
	wire  [1:0]   bresp;
	wire          bvalid;
	reg           bready = 1;
	reg  [AW-1:0] araddr = 0;
	reg           arvalid = 0;
	wire          arready;
	wire  [31:0]  rdata;
	wire  [1:0]   rresp;
	wire          rvalid;
	reg           rready = 1;

	wire [6:0]  idelay_en;
	wire [4:0]  idelay_tap;
	wire [11:0] phase_inc;
	wire        phy_mode;

	integer pass_cnt = 0, fail_cnt = 0;

	task chk;
		input [255:0] name;
		input         cond;
		begin
			if (cond) begin pass_cnt = pass_cnt + 1; $display("PASS %0s", name); end
			else      begin fail_cnt = fail_cnt + 1; $display("FAIL %0s", name); end
		end
	endtask

	always #(ACLK_NS/2.0) aclk = ~aclk;

	// DATA_CLK: stopped (held low) when dclk_en=0, otherwise toggling.
	initial forever begin
		#(DCLK_NS/2.0);
		data_clk = dclk_en ? ~data_clk : 1'b0;
	end

	axi_ctrl_v1_0 #(.C_S00_AXI_DATA_WIDTH(32), .C_S00_AXI_ADDR_WIDTH(AW)) dut (
		.idelay_en(idelay_en), .idelay_tap(idelay_tap),
		.phase_inc(phase_inc), .phy_mode(phy_mode),
		.data_clk(data_clk), .rst_source_ok(rst_source_ok),
		.s00_axi_aclk(aclk), .s00_axi_aresetn(aresetn),
		.s00_axi_awaddr(awaddr), .s00_axi_awprot(3'b0),
		.s00_axi_awvalid(awvalid), .s00_axi_awready(awready),
		.s00_axi_wdata(wdata), .s00_axi_wstrb(wstrb),
		.s00_axi_wvalid(wvalid), .s00_axi_wready(wready),
		.s00_axi_bresp(bresp), .s00_axi_bvalid(bvalid), .s00_axi_bready(bready),
		.s00_axi_araddr(araddr), .s00_axi_arprot(3'b0),
		.s00_axi_arvalid(arvalid), .s00_axi_arready(arready),
		.s00_axi_rdata(rdata), .s00_axi_rresp(rresp),
		.s00_axi_rvalid(rvalid), .s00_axi_rready(rready)
	);

	task axi_write;
		input [AW-1:0] addr;
		input [31:0]   data;
		begin
			@(posedge aclk);
			awaddr <= addr; wdata <= data; awvalid <= 1'b1; wvalid <= 1'b1;
			@(posedge aclk);
			while (!(awready && wready)) @(posedge aclk);
			awvalid <= 1'b0; wvalid <= 1'b0;
			@(posedge aclk);
			while (!bvalid) @(posedge aclk);
		end
	endtask

	task axi_read;
		input  [AW-1:0] addr;
		output [31:0]   data;
		begin
			@(posedge aclk);
			araddr <= addr; arvalid <= 1'b1;
			@(posedge aclk);
			while (!arready) @(posedge aclk);
			arvalid <= 1'b0;
			@(posedge aclk);
			while (!rvalid) @(posedge aclk);
			data = rdata;
		end
	endtask

	// STATUS bit fields (offset 0x10)
	function [3:0]  f_wlog2;  input [31:0] w; begin f_wlog2  = w[27:24]; end endfunction
	function [15:0] f_cycles; input [31:0] w; begin f_cycles = w[23:8];  end endfunction
	function [3:0]  f_seq;    input [31:0] w; begin f_seq    = w[7:4];   end endfunction
	function        f_active; input [31:0] w; begin f_active = w[3];     end endfunction
	function        f_rso;    input [31:0] w; begin f_rso    = w[2];     end endfunction
	function        f_done;   input [31:0] w; begin f_done   = w[1];     end endfunction
	function        f_seen;   input [31:0] w; begin f_seen   = w[0];     end endfunction

	// Wait until win_seq changes, then return the freshly completed window.
	task wait_new_window;
		output [31:0] st;
		reg [31:0] s0, s1;
		integer    guard;
		begin
			axi_read(6'h10, s0);
			guard = 0;
			s1 = s0;
			while (f_seq(s1) == f_seq(s0) && guard < 200000) begin
				axi_read(6'h10, s1);
				guard = guard + 1;
			end
			if (guard >= 200000) $display("WARN wait_new_window timed out");
			st = s1;
		end
	endtask

	reg [31:0] st, st2;

	initial begin
		$display("=== axi_ctrl_tb ===");

		aresetn = 1'b0;
		dclk_en = 1'b0;             // DATA_CLK absent
		rst_source_ok = 1'b0;
		repeat (5) @(posedge aclk);
		aresetn = 1'b1;
		repeat (2) @(posedge aclk);

		// ---------------------------------------------------------------
		// Group 1 -- NEGATIVE criterion: no DATA_CLK => zero, not "unknown".
		// ---------------------------------------------------------------
		wait_new_window(st);
		chk("dead: win_cycles == 0",        f_cycles(st) == 16'd0);
		chk("dead: clk_seen == 0",          f_seen(st)   == 1'b0);
		chk("dead: rst_source_ok == 0",     f_rso(st)    == 1'b0);
		chk("dead: WIN_LOG2 field == 15",   f_wlog2(st)  == 4'd15);
		chk("dead: window FSM still runs",  f_seq(st)    != f_seq(32'h0) || f_done(st) == 1'b1);

		// ---------------------------------------------------------------
		// Group 2 -- POSITIVE criterion: DATA_CLK at 30 MHz.
		// ---------------------------------------------------------------
		dclk_en = 1'b1;
		rst_source_ok = 1'b1;
		wait_new_window(st);
		wait_new_window(st);        // discard the first (partial) window
		chk("alive: clk_seen == 1",         f_seen(st) == 1'b1);
		chk("alive: rst_source_ok == 1",    f_rso(st)  == 1'b1);
		chk("alive: win_cycles in 19660 +/- 2",
		    f_cycles(st) >= 16'd19658 && f_cycles(st) <= 16'd19663);
		chk("alive: win_active low at read", f_active(st) == 1'b0);
		chk("alive: win_done == 1",          f_done(st)   == 1'b1);

		// ---------------------------------------------------------------
		// Group 3 -- slv_reg0..3 must not regress.
		// ---------------------------------------------------------------
		axi_write(6'h00, 32'h0000_007F);
		axi_write(6'h04, 32'h0000_0015);
		axi_write(6'h08, 32'h0000_0ABC);
		axi_write(6'h0C, 32'h0000_0001);
		axi_read(6'h00, st); chk("reg0 == 0x7F",  st[6:0] == 7'h7F);
		axi_read(6'h04, st); chk("reg1 == 0x15",  st[4:0] == 5'h15);
		axi_read(6'h08, st); chk("reg2 == 0xABC", st[11:0] == 12'hABC);
		axi_read(6'h0C, st); chk("reg3 == 1",     st[0] == 1'b1);

		// ---------------------------------------------------------------
		// Group 4 -- STATUS must be READ-ONLY: writing 0x10 changes nothing.
		// ---------------------------------------------------------------
		axi_write(6'h10, 32'hFFFF_FFFF);
		wait_new_window(st);
		chk("read-only: WIN_LOG2 still 15",  f_wlog2(st) == 4'd15);
		chk("read-only: cycles still sane",
		    f_cycles(st) >= 16'd19658 && f_cycles(st) <= 16'd19663);
		chk("read-only: reserved bits zero", st[31:28] == 4'h0);
		// win_seq must keep rolling forward (proves the FSM was not clobbered)
		axi_read(6'h10, st2);
		chk("read-only: seq still rolling",  f_seq(st2) != 4'h0 || f_done(st2) == 1'b1);

		// ---------------------------------------------------------------
		// Group 5 -- rst_source_ok is a synchronised LEVEL, not a latch.
		// ---------------------------------------------------------------
		rst_source_ok = 1'b0;
		wait_new_window(st);
		wait_new_window(st);
		chk("rso: follows low",  f_rso(st) == 1'b0);
		rst_source_ok = 1'b1;
		wait_new_window(st);
		wait_new_window(st);
		chk("rso: follows high", f_rso(st) == 1'b1);

		$display("=== %0d PASS, %0d FAIL ===", pass_cnt, fail_cnt);
		if (fail_cnt == 0) $display("PASS axi_ctrl_tb");
		$finish;
	end

	initial begin
		#20_000_000;
		$display("FAIL timeout");
		$finish;
	end

endmodule
```

- [ ] **Step 6: 写 STATUS 寄存器的用户逻辑**

替换 `axi_ctrl_v1_0_S00_AXI.v` 末尾的 `// Add user logic here` 块
（原第 531–536 行）为：

```verilog
	// Add user logic here
	assign  idelay_en  = slv_reg0[6:0];
	assign  idelay_tap = slv_reg1[4:0];
	assign  phase_inc  = slv_reg2[11:0];
	assign  phy_mode   = slv_reg3[0];

	// ---------------------------------------------------------------------
	// Read-only STATUS (offset 0x10).
	//
	// Answers "is the AD9361's DATA_CLK actually reaching the FPGA, and how
	// fast?" without any software timing.  A DATA_CLK-domain Gray counter is
	// sampled by a self-running window on FCLK0; software reads the count
	// together with a rolling sequence number, so it never reads a torn
	// window.
	//
	// The counter MUST be clocked by data_clk, not sampled by FCLK0: FCLK0 is
	// 50 MHz and the candidate DATA_CLK rates are 30/60 MHz, so sampling would
	// alias (drop edges) while still reading something plausible.  Gray coding
	// only makes the multi-bit crossing safe -- it does not recover lost edges.
	// ---------------------------------------------------------------------

	// DATA_CLK-domain free-running counter.  Reset is async-asserted from the
	// AXI reset (so the value is defined even when data_clk never ticks) and
	// sync-released into the DATA_CLK domain by the 2-FF chain below.
	reg [1:0] dclk_rst_sync;
	always @(posedge data_clk or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) dclk_rst_sync <= 2'b00;
		else                dclk_rst_sync <= {dclk_rst_sync[0], 1'b1};
	end
	wire dclk_rst_n = dclk_rst_sync[1];

	reg [15:0] dclk_bin;
	reg [15:0] dclk_gray;
	always @(posedge data_clk or negedge dclk_rst_n) begin
		if (!dclk_rst_n) begin
			dclk_bin  <= 16'd0;
			dclk_gray <= 16'd0;
		end else begin
			dclk_bin  <= dclk_bin + 16'd1;
			dclk_gray <= (dclk_bin + 16'd1) ^ ((dclk_bin + 16'd1) >> 1);
		end
	end

	// 3-deep sync of the Gray word into FCLK0.  Any sample is the old or the
	// new value, never a mix -- that is what Gray coding buys.
	reg [15:0] gray_s0, gray_s1, gray_s2;
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) {gray_s0, gray_s1, gray_s2} <= 48'd0;
		else                {gray_s0, gray_s1, gray_s2} <= {dclk_gray, gray_s0, gray_s1};
	end

	function [15:0] gray2bin;
		input [15:0] g;
		integer i;
		begin
			gray2bin[15] = g[15];
			for (i = 14; i >= 0; i = i - 1)
				gray2bin[i] = gray2bin[i+1] ^ g[i];
		end
	endfunction

	wire [15:0] dclk_now     = gray2bin(gray_s2);
	// A Gray increment changes exactly one bit, so "s1 != s2" means an edge
	// arrived -- this is the liveness indicator, not a data comparison.
	wire        gray_changed = (gray_s1 != gray_s2);

	// Self-running measurement window: 2^WIN_LOG2 FCLK0 cycles = 655.36 us.
	//
	// 勘误（2026-09-17 执行时发现，**这是原计划的一个真 bug**）：
	// 下面这个 FSM 里 `win_done=1 && win_active=0` **只持续 1 个 FCLK0 周期（20 ns）**
	// —— 发布那一拍之后 `!win_active` 立刻把下一窗重启了。而 TB 的
	// `wait_new_window` 在 `win_seq` 变化后**用一次 AXI 读**取状态（读本身跨若干周期），
	// 于是必然读到 `win_active=1, win_done=0`，两条判据
	// （`alive: win_active low at read` / `alive: win_done == 1`）**必定 FAIL**。
	// 这不是 TB 太苛刻：软件侧 `status_read_fresh()` 也依赖同样的"读到的是一份稳定结果"，
	// 所以这是**判据对实现的正当要求**，实现该改。
	//
	// 修法：加一个 WIN_GAP 发布期（见下面的实测代码），发布态**保持**若干周期再开下一窗。
	// 关键约束：**测量本身仍是精确 2^WIN_LOG2 个 FCLK0 周期**，WIN_GAP 只延长两次测量
	// 之间的间隔，不进 `win_cycles` 的账。若把 gap 算进去，`f_DATA_CLK` 就会系统性偏低。
	localparam [3:0]  WIN_LOG2   = 4'd15;
	localparam [15:0] WIN_CYCLES = (16'd1 << WIN_LOG2) - 16'd1;
	localparam [15:0] WIN_GAP    = 16'd255;

	reg [15:0] win_cnt;
	reg        win_active;
	reg [3:0]  win_seq;
	reg        win_done;
	reg [15:0] win_base;
	reg [15:0] win_cycles;
	reg        clk_seen;
	reg [15:0] win_gap_cnt;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			win_cnt    <= 16'd0;
			win_active <= 1'b0;
			win_seq    <= 4'd0;
			win_done   <= 1'b0;
			win_base   <= 16'd0;
			win_cycles <= 16'd0;
			clk_seen   <= 1'b0;
			win_gap_cnt <= 16'd0;
		end else begin
			if (gray_changed) clk_seen <= 1'b1;
			if (!win_active && !win_done) begin
				// first window after reset
				win_active <= 1'b1;
				win_base   <= dclk_now;
				win_cnt    <= 16'd0;
				win_done   <= 1'b0;
				win_gap_cnt <= 16'd0;
			end else if (win_active) begin
				if (win_cnt == WIN_CYCLES) begin
					win_cycles <= dclk_now - win_base;
					win_seq    <= win_seq + 4'd1;
					win_done   <= 1'b1;
					win_active <= 1'b0;
					win_gap_cnt <= 16'd0;
				end else begin
					win_cnt <= win_cnt + 16'd1;
				end
			end else begin
				// publish period: results held, next window deferred by WIN_GAP
				if (win_gap_cnt == WIN_GAP) begin
					win_active <= 1'b1;
					win_base   <= dclk_now;
					win_cnt    <= 16'd0;
					win_done   <= 1'b0;
					win_gap_cnt <= 16'd0;
				end else begin
					win_gap_cnt <= win_gap_cnt + 16'd1;
				end
			end
		end
	end
	// rst_source_ok: a DATA_CLK-domain level.  It means "rf_tx_source's own
	// reset chain has filled", NOT "the phy is ready" -- see the plan's D1.
	reg [1:0] rso_sync;
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) rso_sync <= 2'b00;
		else                rso_sync <= {rso_sync[0], rst_source_ok};
	end

	assign status_word = { 4'h0,          // [31:28] reserved
	                       WIN_LOG2,      // [27:24] window is 2^15 FCLK0 cycles
	                       win_cycles,    // [23:8]  DATA_CLK edges in the window
	                       win_seq,       // [7:4]   rolling window index
	                       win_active,    // [3]     1 = window in progress
	                       rso_sync[1],   // [2]     rst_source_ok (synced level)
	                       win_done,      // [1]     last window completed
	                       clk_seen };    // [0]     sticky: an edge was seen
	// User logic ends
```

```make
$(OBJDIR)/%.o: vendor/%.c | $(OBJDIR)
```

```c
/* platform_rf.h -- the single source of truth for addresses, pin order and the
 * handful of shapes the vendor driver expects.  Nothing else in board_app may
 * hardcode an address.
 */
#ifndef PLATFORM_RF_H_
#define PLATFORM_RF_H_

#include <stdint.h>

/* ---- PL apertures.  Taken from rf_overlay/out/rf_overlay.hwh; addresses move
 * when the BD changes, so re-read the .hwh after every rebuild. ---- */
#define RF_AXI_CTRL_BASE   0x40000000u   /* axi_ctrl_0     */
#define RF_PL_SPI_BASE     0x40001000u   /* pl_spi_0       */
#define RF_GPIO_CTRL_BASE  0x41200000u   /* axi_gpio_ctrl  */

/* Every register access is an ABSOLUTE address, composed here.  It has to be
 * absolute rather than "base plus offset" passed around separately: three
 * different apertures each have a register at offset 0x00 (IDELAY_EN, SPI_CTRL,
 * GPIO_DATA), so a bare offset does not identify a register.  On the board this
 * lets one mmap cache serve all three apertures, exactly like spi_probe.py's
 * Mmio does; in the host model it lets fake_pl.c dispatch on one integer. */
#define RF_CTRL_A(o)   (RF_AXI_CTRL_BASE  + (o))
#define RF_SPI_A(o)    (RF_PL_SPI_BASE    + (o))
#define RF_GPIO_A(o)   (RF_GPIO_CTRL_BASE + (o))

/* axi_ctrl_0 register offsets (word addresses) */
#define RF_REG_IDELAY_EN   0x00u
#define RF_REG_IDELAY_TAP  0x04u
#define RF_REG_PHASE_INC   0x08u
#define RF_REG_PHY_MODE    0x0Cu
#define RF_REG_STATUS      0x10u   /* read-only, see the plan's bit table */

/* pl_spi_master register offsets (word addresses) */
#define RF_SPI_CTRL        0x00u   /* [7:0] N_TX, [15:8] N_RX, [16] START */
#define RF_SPI_INSTR       0x04u
#define RF_SPI_TXD0        0x08u
#define RF_SPI_TXD1        0x0Cu
#define RF_SPI_RXD0        0x10u
#define RF_SPI_RXD1        0x14u
#define RF_SPI_STATUS      0x18u   /* [0] BUSY, [1] DONE (sticky) */
#define RF_SPI_DIV         0x1Cu   /* SCLK half-period in aclk cycles */

/* axi_gpio_ctrl (9 outputs, all output).  Bit order matches the BD and also
 * matches the reference main.c's EMIO numbering, so pin number == bit number. */
#define RF_GPIO_DATA       0x00u
#define RF_GPIO_TRI        0x04u

#define RF_GPIO_TXNRX      0
#define RF_GPIO_ENABLE     1
#define RF_GPIO_RESETB     2
#define RF_GPIO_SYNC       3
#define RF_GPIO_EN_AGC     4
#define RF_GPIO_CTL0       5
#define RF_GPIO_CTL1       6
#define RF_GPIO_CTL2       7
#define RF_GPIO_CTL3       8
#define RF_GPIO_COUNT      9

/* SCLK half-period in 50 MHz aclk cycles.  10 -> 2.5 MHz SCLK, the value the
 * already-passing Python probe uses (rf_overlay/sw/spi_probe.py). */
#define RF_SPI_DIV_DEFAULT 10u

/* ---- PL STATUS word decode (offset RF_REG_STATUS) ---- */
#define RF_STATUS_CLK_SEEN(w)    (((w) >>  0) & 0x1u)
#define RF_STATUS_WIN_DONE(w)    (((w) >>  1) & 0x1u)
#define RF_STATUS_RSO(w)         (((w) >>  2) & 0x1u)
#define RF_STATUS_WIN_ACTIVE(w)  (((w) >>  3) & 0x1u)
#define RF_STATUS_WIN_SEQ(w)     (((w) >>  4) & 0xFu)
#define RF_STATUS_WIN_CYCLES(w)  (((w) >>  8) & 0xFFFFu)
#define RF_STATUS_WIN_LOG2(w)    (((w) >> 24) & 0xFu)

/* f_FCLK0 = 50 MHz, window = 2^WIN_LOG2 cycles -> f_DATA_CLK in Hz. */
#define RF_STATUS_F_DATA_CLK_HZ(w) \
	(((uint64_t)RF_STATUS_WIN_CYCLES(w) * 50000000ull) >> RF_STATUS_WIN_LOG2(w))

/* ---- axi_ctrl_0 accessors.  These are ours, not the vendor driver's; they live
 * in platform_rf.c so that the aperture bases appear in exactly one .c file. ---- */
uint32_t rf_reg_read(uint32_t off);
void     rf_reg_write(uint32_t off, uint32_t val);

/* ---- trace / audit ---- */
void rf_trace_open(const char *path);
int  rf_trace_flush(void);   /* returns records written, or -1 */
void rf_trace_overflow_report(void);

#endif /* PLATFORM_RF_H_ */
```

```c
/* fake_pl.c -- a C model of the PL registers AND of the AD9361's SPI decode.
 *
 * The PL half mirrors rf_overlay/rtl/pl_spi_master.v: one CSN-low burst of
 * 2 + N_TX + N_RX bytes, byte0 = INSTR[15:8], DONE sticky until the next START.
 *
 * The device half is the part that matters.  The PL master does NOT interpret
 * the CNT field -- it just clocks bytes -- so a send-side-only model cannot
 * catch an encoding error.  This models the AD9361's rule: a write must carry
 * exactly CNT+1 data bytes or the part drops the whole command, and a
 * multi-byte write takes the LAST data byte.
 */
#include "fake_pl.h"
#include "platform_rf.h"
#include <string.h>
#include <stdio.h>

int     fake_spi_rejected;
int     fake_spi_writes;
int     fake_last_burst_len;
int     fake_last_ntx, fake_last_nrx;
uint8_t fake_reg[0x400];

static uint32_t spi_ctrl, spi_instr, spi_txd0, spi_txd1, spi_rxd0, spi_rxd1;
static uint32_t spi_status, spi_div;
static uint32_t gpio_data, gpio_tri;
static uint32_t ctrl_idelay_en, ctrl_idelay_tap, ctrl_phase_inc, ctrl_phy_mode;
static uint32_t status_word = 0;    /* the read-only PL STATUS register */
static uint8_t  product_id = 0x0Au;

void fake_pl_set_product_id(uint8_t v) { product_id = v; }
void fake_pl_set_status(uint32_t v)    { status_word = v; }

void fake_pl_reset(void)
{
	spi_ctrl = spi_instr = spi_txd0 = spi_txd1 = 0;
	spi_rxd0 = spi_rxd1 = spi_status = 0;
	spi_div = 25;                       /* pl_spi_master's reset value */
	gpio_data = gpio_tri = 0;
	ctrl_idelay_en = ctrl_idelay_tap = ctrl_phase_inc = ctrl_phy_mode = 0;
	status_word = 0;                    /* no DATA_CLK: the dead-clock state */
	fake_spi_rejected = fake_spi_writes = 0;
	fake_last_burst_len = 0;
	fake_last_ntx = fake_last_nrx = 0;
	memset(fake_reg, 0, sizeof(fake_reg));
	fake_reg[0x037] = product_id;       /* REG_PRODUCT_ID */
}

/* One burst.  Mirrors the PL master's byte order exactly.
 *
 * 勘误（2026-09-17 执行时发现，**这是原计划的一个真 bug**）：
 * `n_tx` 是 **数据字节数，不含那 2 个指令字节**。依据是真实 RTL：
 *   - `pl_spi_master.v:305`  `n_tx <= s_axi_wdata[7:0]` —— CTRL[7:0] 存的是
 *     `spi_write_then_read` 传下来的 `n_data`（= `n_tx - 2`）；
 *   - `pl_spi_master.v:189`  `total_bytes = 5'd2 + lat_n_tx + lat_n_rx` ——
 *     **RTL 自己把那 2 个指令字节加上去**。
 * 所以 `n_tx` 已经是"数据字节"，与 `2 + n_tx + n_rx` 是两回事。
 *
 * 原计划在下面两处**重复加了那 2 个字节**，后果是**全盘误判**：
 *   (1) 拒绝判据写成 `n_tx < 2 + need` —— 单字节写（n_tx=1, need=1）会算出
 *       `1 < 3` 为真，于是**每一条合法写都被判为"被器件拒绝"**；
 *   (2) `fake_last_burst_len = n_tx + n_rx` 漏了那 2 个字节，读事务报 1 而不是 3。
 *
 * **最阴的地方**：反向用例（CNT=1 必须被拒）会**因为错误的原因通过** ——
 * 合法写和非法写一起被拒，测试看着是绿的却什么都没证明。这正是本任务要防的
 * "判据退化成恒真"。所以这个 bug 必须修，不能靠调测试迁就。
 */
static void fake_burst(uint32_t n_data_tx, uint32_t n_rx)
{
	uint8_t tx[24], rx[16];
	uint32_t i;
	uint32_t n_tx_total = 2u + n_data_tx;   /* what the wire actually carries */
	uint16_t cmd;
	uint32_t cnt, addr, is_write;

	fake_last_ntx = (int)n_data_tx;
	fake_last_nrx = (int)n_rx;
	/* The burst on the wire is instruction (2 bytes) + data + received bytes --
	 * this is what pl_spi_master.v:189 computes as total_bytes. */
	fake_last_burst_len = (int)(n_tx_total + n_rx);

	/* byte 0/1 are the instruction word; data bytes follow.  The PL shifts out
	 * txd0[7:0] first (tx_byte = txd_full[(byte_cnt-2)*8 +: 8]), i.e. LSB first,
	 * which is what the packing below reproduces. */
	tx[0] = (uint8_t)(spi_instr >> 8);
	tx[1] = (uint8_t)(spi_instr & 0xFF);
	tx[2] = (uint8_t)(spi_txd0 & 0xFF);
	tx[3] = (uint8_t)((spi_txd0 >> 8) & 0xFF);
	tx[4] = (uint8_t)((spi_txd0 >> 16) & 0xFF);
	tx[5] = (uint8_t)((spi_txd0 >> 24) & 0xFF);
	tx[6] = (uint8_t)(spi_txd1 & 0xFF);
	tx[7] = (uint8_t)((spi_txd1 >> 8) & 0xFF);
	for (i = 8; i < sizeof(tx); i++) tx[i] = 0;
	for (i = 0; i < sizeof(rx); i++) rx[i] = 0;

	cmd      = (uint16_t)((tx[0] << 8) | tx[1]);
	is_write = (cmd >> 15) & 0x1u;
	cnt      = (cmd >> 12) & 0x7u;      /* the field as transmitted */
	addr     = cmd & 0x3FFu;

	/* The DEVICE's rule: the field is byte-count-minus-one, so the command
	 * claims cnt+1 data bytes.  A write whose burst does not carry exactly
	 * that many is silently discarded -- which is precisely how the CNT=1
	 * encoding bug hid: reads encode 0 under either rule, so only writes
	 * died. */
	if (is_write) {
		uint32_t need = cnt + 1u;       /* data bytes the command claims */
		if (n_data_tx < need) {
			fake_spi_rejected++;
			/* device drops it; register file unchanged */
		} else if (addr < 0x400u) {
			fake_reg[addr] = tx[2u + need - 1u];   /* LAST byte */
			fake_spi_writes++;
		}
	} else {
		/* read: the data bytes come back from the register file, starting at
		 * the addressed register and auto-incrementing, as the part does. */
		for (i = 0; i < n_rx && i < sizeof(rx); i++)
			rx[i] = (addr + i < 0x400u) ? fake_reg[addr + i] : 0;
		if (addr == 0x037u) rx[0] = product_id;
	}

	spi_rxd0 = (uint32_t)rx[0]
	         | ((uint32_t)rx[1] << 8)
	         | ((uint32_t)rx[2] << 16)
	         | ((uint32_t)rx[3] << 24);
	spi_rxd1 = (uint32_t)rx[4]
	         | ((uint32_t)rx[5] << 8)
	         | ((uint32_t)rx[6] << 16)
	         | ((uint32_t)rx[7] << 24);
	spi_status |= 0x2u;                 /* DONE */
}

uint32_t fake_pl_read(uint32_t off)
{
	switch (off) {
	case RF_SPI_A(RF_SPI_CTRL):   return spi_ctrl & ~0x10000u;
	case RF_SPI_A(RF_SPI_INSTR):  return spi_instr;
	case RF_SPI_A(RF_SPI_TXD0):   return spi_txd0;
	case RF_SPI_A(RF_SPI_TXD1):   return spi_txd1;
	case RF_SPI_A(RF_SPI_RXD0):   return spi_rxd0;
	case RF_SPI_A(RF_SPI_RXD1):   return spi_rxd1;
	case RF_SPI_A(RF_SPI_STATUS): return spi_status;
	case RF_SPI_A(RF_SPI_DIV):    return spi_div;
	case RF_GPIO_A(RF_GPIO_DATA): return gpio_data;
	case RF_GPIO_A(RF_GPIO_TRI):  return gpio_tri;
	/* 勘误（2026-09-17 执行时修正）：这四个 read case 原计划漏了。
	 * fake_pl_write 存了它们，读却落进 default 返回 0xDEADBEEF —— 于是
	 * "phy_mode round-trips" 与 "idelay_en/tap/phase_inc round-trip" 两条
	 * 判据必然失败。读侧必须与写侧镜像。 */
	case RF_CTRL_A(RF_REG_IDELAY_EN):  return ctrl_idelay_en;
	case RF_CTRL_A(RF_REG_IDELAY_TAP): return ctrl_idelay_tap;
	case RF_CTRL_A(RF_REG_PHASE_INC):  return ctrl_phase_inc;
	case RF_CTRL_A(RF_REG_PHY_MODE):   return ctrl_phy_mode;
	/* STATUS is deliberately read-only: it has a read case and NO write case. */
	case RF_CTRL_A(RF_REG_STATUS): return status_word;
	default:                      return 0xDEADBEEFu;  /* unmapped, as on the board */
	}
}

void fake_pl_write(uint32_t off, uint32_t val)
{
	switch (off) {
	case RF_SPI_A(RF_SPI_CTRL):
		spi_status &= ~0x2u;            /* DONE clears on the next START */
		spi_ctrl = val;
		if (val & 0x10000u) {
			uint32_t n_tx = val & 0xFFu;
			uint32_t n_rx = (val >> 8) & 0xFFu;
			fake_burst(n_tx, n_rx);
		}
		break;
	case RF_SPI_A(RF_SPI_INSTR):  spi_instr = val; break;
	case RF_SPI_A(RF_SPI_TXD0):   spi_txd0 = val;  break;
	case RF_SPI_A(RF_SPI_TXD1):   spi_txd1 = val;  break;
	case RF_SPI_A(RF_SPI_DIV):    spi_div = val;   break;
	case RF_GPIO_A(RF_GPIO_DATA): gpio_data = val & 0x1FFu; break;
	case RF_GPIO_A(RF_GPIO_TRI):  gpio_tri = val;  break;
	/* axi_ctrl_0's config registers, so rf_main's `status` subcommand and the
	 * init sequence can be exercised on the host too. */
	case RF_CTRL_A(RF_REG_IDELAY_EN):  ctrl_idelay_en  = val; break;
	case RF_CTRL_A(RF_REG_IDELAY_TAP): ctrl_idelay_tap = val; break;
	case RF_CTRL_A(RF_REG_PHASE_INC):  ctrl_phase_inc  = val; break;
	case RF_CTRL_A(RF_REG_PHY_MODE):   ctrl_phy_mode   = val; break;
	/* RF_REG_STATUS is deliberately absent: it is read-only, and a write that
	 * lands nowhere is exactly what the board does. */
	default: break;
	}
}
```

```c
static int pass_cnt, fail_cnt;
static void chk(const char *name, int cond)
{
	if (cond) { pass_cnt++; printf("PASS %s\n", name); }
	else      { fail_cnt++; printf("FAIL %s\n", name); }

/* ad9361.h already defines both (same values); guard to avoid -Wmacro-redefined. */
#ifndef REG_PRODUCT_ID
#define REG_PRODUCT_ID 0x037u
#endif
#ifndef REG_CTRL
#define REG_CTRL       0x3DFu
#endif

static struct spi_device spi;

static uint8_t reg_read(uint32_t addr)
{
	uint8_t tx[2], rx[1];
	uint16_t cmd = AD_READ | AD_CNT(1) | AD_ADDR(addr);
	tx[0] = (uint8_t)(cmd >> 8);
	tx[1] = (uint8_t)(cmd & 0xFF);
	memset(rx, 0, sizeof(rx));
	spi_write_then_read(&spi, tx, 2, rx, 1);
	return rx[0];
}

	uint16_t cmd = AD_WRITE | AD_CNT(1) | AD_ADDR(addr);
	tx[0] = (uint8_t)(cmd >> 8);
	tx[1] = (uint8_t)(cmd & 0xFF);
	tx[2] = val;
	spi_write_then_read(&spi, tx, 3, NULL, 0);
}
```

```c
/* platform_rf.c -- the vendor driver's platform layer, mapped onto this design.
 *
 * Only these functions are ours; every vendor file in vendor/ is verbatim.  The
 * reference platform.c cannot be reused because it drives XSpiPs and puts the
 * nine control pins on PS EMIO GPIO, whereas here they are a 9-bit PL aperture
 * and the SPI port is our own PL master.
 *
 * With -DPLATFORM_HOST the MMIO goes to test/fake_pl.c instead of /dev/mem, so
 * the exact same source runs both on the board and in the host test.
 */
#include "platform_rf.h"
#include "platform.h"

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#ifdef PLATFORM_HOST
#include "test/fake_pl.h"
#define MMIO_READ(a)    fake_pl_read((a))
#define MMIO_WRITE(a,v) fake_pl_write((a),(v))
#else
#include <fcntl.h>
#include <unistd.h>
#include <time.h>
#include <sys/mman.h>

/* One mmap cache, keyed by page base, serving all three apertures -- the same
 * shape spi_probe.py's Mmio uses.  Access is by ABSOLUTE address so that the
 * host model can be a single function; see platform_rf.h for why. */
#define RF_PAGE_SIZE 4096u
#define RF_MAX_MAPS  8

static int      mem_fd = -1;
static uint8_t *map_base[RF_MAX_MAPS];
static uint32_t map_addr[RF_MAX_MAPS];
static int      map_count;

static uint8_t *map_region(uint32_t addr)
{
	uint32_t base = addr & ~(RF_PAGE_SIZE - 1u);
	int i;

	for (i = 0; i < map_count; i++)
		if (map_addr[i] == base) return map_base[i];

	if (map_count >= RF_MAX_MAPS) {
		fprintf(stderr, "too many mmap regions\n");
		exit(2);
	}
	if (mem_fd < 0) {
		mem_fd = open("/dev/mem", O_RDWR | O_SYNC);
		if (mem_fd < 0) { perror("open /dev/mem"); exit(2); }
	}
	map_base[map_count] = (uint8_t *)mmap(NULL, RF_PAGE_SIZE,
	                                      PROT_READ | PROT_WRITE, MAP_SHARED,
	                                      mem_fd, base);
	if (map_base[map_count] == MAP_FAILED) {
		fprintf(stderr, "mmap 0x%08X failed\n", base);
		exit(2);
	}
	map_addr[map_count] = base;
	return map_base[map_count++];
}

static uint32_t mmio_read(uint32_t addr)
{
	uint8_t *p = map_region(addr);
	return *(volatile uint32_t *)(p + (addr & (RF_PAGE_SIZE - 1u)));
}

static void mmio_write(uint32_t addr, uint32_t v)
{
	uint8_t *p = map_region(addr);
	*(volatile uint32_t *)(p + (addr & (RF_PAGE_SIZE - 1u))) = v;
}

#define MMIO_READ(a)     mmio_read((a))
#define MMIO_WRITE(a,v)  mmio_write((a),(v))
#endif /* PLATFORM_HOST */

/* ------------------------------------------------------------------ timebase
 *
 * udelay/mdelay busy-wait on CLOCK_MONOTONIC rather than calling nanosleep.
 * The driver uses them for short polls inside the VCO calibration sequence,
 * where a scheduler round-trip would both stretch init and risk missing a
 * timing window; a spin costs a few hundred microseconds of CPU and nothing
 * else.  msleep_interruptible is only used for 100 ms and 1 ms waits outside
 * any calibration window, so it can afford to sleep.
 */
/* 勘误（2026-09-17 执行时修正）：now_ns 的 #ifdef 要包住**整个函数定义**，
 * 而不是只包函数体。原写法在 host 上让 now_ns 有定义却没人调用，触发
 * -Wunused-function。host 上 udelay 是空操作，根本不需要 now_ns。 */
#ifndef PLATFORM_HOST
static uint64_t now_ns(void)
{
	struct timespec ts;
	clock_gettime(CLOCK_MONOTONIC, &ts);
	return (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
}
#endif

void udelay(unsigned long usecs)
{
#ifdef PLATFORM_HOST
	/* The host model's "hardware" is instantaneous, so the delays exist only
	 * to exercise the call paths.  Burning real CPU here would make the test
	 * suite take minutes for nothing -- so this whole function is a no-op on
	 * the host, and now_ns() is not even compiled in. */
	(void)usecs;
#else
	uint64_t end = now_ns() + (uint64_t)usecs * 1000ull;
	while (now_ns() < end)
		;
#endif
}

void mdelay(unsigned long msecs)
{
	udelay(msecs * 1000ul);
}

unsigned long msleep_interruptible(unsigned int msecs)
{
#ifdef PLATFORM_HOST
	(void)msecs;
	return 0;
#else
	struct timespec ts;
	ts.tv_sec  = (time_t)(msecs / 1000u);
	ts.tv_nsec = (long)(msecs % 1000u) * 1000000L;
	nanosleep(&ts, NULL);
	return 0;
#endif
}

/* ------------------------------------------------------------- axi_ctrl_0
 *
 * Our own registers, not the vendor's.  Exposed through platform_rf.h so that
 * rf_main.c and selftest.c never spell an aperture base themselves.
 */
uint32_t rf_reg_read(uint32_t off)
{
	return MMIO_READ(RF_CTRL_A(off));
}

void rf_reg_write(uint32_t off, uint32_t val)
{
	MMIO_WRITE(RF_CTRL_A(off), val);

/* ------------------------------------------------------------------ trace
 *
 * Always on, and deliberately NOT a file write per transaction: a write() in
 * the middle of the VCO calibration's timing window would perturb exactly what
 * this round is trying to measure.  Records go to a fixed in-memory buffer and
 * are flushed once, after init has finished.
 */
#define RF_TRACE_CAP  (256 * 1024)     /* records; ~12 bytes each = ~3 MiB */
typedef struct {
	uint8_t  rw;          /* 0 = read, 1 = write */
	uint16_t addr;
	uint8_t  n;
	uint8_t  data[8];
} rf_trace_rec;

static rf_trace_rec *trace_buf;
static uint32_t      trace_len;
static uint32_t      trace_lost;
static const char   *trace_path;

static void trace_add(int rw, uint16_t addr, const uint8_t *d, unsigned n)
{
	rf_trace_rec *r;
	if (!trace_buf) return;
	if (trace_len >= RF_TRACE_CAP) { trace_lost++; return; }
	r = &trace_buf[trace_len++];
	r->rw   = (uint8_t)rw;
	r->addr = addr;
	r->n    = (uint8_t)(n > 8 ? 8 : n);
	if (d && r->n) memcpy(r->data, d, r->n);
}

void rf_trace_open(const char *path)
{
	trace_path = path;
	if (!trace_buf)
		trace_buf = (rf_trace_rec *)calloc(RF_TRACE_CAP, sizeof(*trace_buf));
	trace_len  = 0;
	trace_lost = 0;
}

/* Written as text, one transaction per line, so the audit pass and the trace
 * can be diffed by eye and by script without a binary parser. */
int rf_trace_flush(void)
{
	FILE *f;
	uint32_t i;

	if (!trace_buf) return -1;
	if (!trace_path) return -1;
	f = fopen(trace_path, "w");
	if (!f) {
		fprintf(stderr, "cannot open trace file %s\n", trace_path);
		return -1;
	}
	for (i = 0; i < trace_len; i++) {
		const rf_trace_rec *r = &trace_buf[i];
		unsigned k;
		fprintf(f, "%c %03X", r->rw ? 'W' : 'R', r->addr);
		for (k = 0; k < r->n; k++)
			fprintf(f, " %02X", r->data[k]);
		fputc('\n', f);
	}
	fclose(f);
	return (int)trace_len;
}

void rf_trace_overflow_report(void)
{
	if (trace_lost)
		fprintf(stderr,
		        "WARNING: trace buffer overflowed, %u transactions lost; "
		        "the audit pass is still complete (it re-reads the device), "
		        "but the write history is not\n", trace_lost);
}

/* --------------------------------------------------------------------- SPI
 *
 * spi_init asserts the mode rather than accepting whatever it is handed.  The
 * PL master is wired for mode 1 (CPOL=0, CPHA=1); if the driver ever asked for
 * another mode the bits would be sampled on the wrong edge and every register
 * read would return plausible garbage, which is the worst possible failure
 * shape.  Failing loudly here costs nothing.
 */
static int spi_ready;

int32_t spi_init(uint32_t device_id, uint8_t clk_pha, uint8_t clk_pol)
{
	(void)device_id;

	if (clk_pha != 1 || clk_pol != 0) {
		fprintf(stderr,
		        "spi_init: PL master is fixed at SPI mode 1 (CPHA=1, CPOL=0), "
		        "asked for CPHA=%u CPOL=%u\n", clk_pha, clk_pol);
		return -1;
	}

	MMIO_WRITE(RF_SPI_A(RF_SPI_DIV), RF_SPI_DIV_DEFAULT);

	/* A wrong base reads the RTL's unmapped default instead of failing, so
	 * probe for it once, here, rather than trusting the first transaction. */
	if (MMIO_READ(RF_SPI_A(RF_SPI_STATUS)) == 0xDEADBEEFu) {
		fprintf(stderr,
		        "spi_init: pl_spi_0 at 0x%08X reads 0xDEADBEEF -- wrong base "
		        "address, or the wrong bitstream is loaded\n", RF_PL_SPI_BASE);
		return -1;
	}

	spi_ready = 1;
	return 0;
}

/* Wait for DONE.  The window is generous because a single mmap read costs tens
 * of microseconds while a 18-byte burst at 2.5 MHz SCLK takes under 60. */
static int spi_wait_done(void)
{
	int i;
	for (i = 0; i < 10000; i++)
		if (MMIO_READ(RF_SPI_A(RF_SPI_STATUS)) & 0x2u)
			return 0;
	fprintf(stderr, "SPI burst timed out (no DONE)\n");
	return -1;
}

int spi_write_then_read(struct spi_device *spi, const unsigned char *txbuf,
                        unsigned n_tx, unsigned char *rxbuf, unsigned n_rx)
{
	uint16_t instr;
	uint32_t n_data, w0 = 0, w1 = 0, i;
	uint64_t full;   /* 64-bit: RXD1 is a second 32-bit read, see the note below */
	uint8_t  rx[8];

	(void)spi;

	if (!spi_ready) {
		fprintf(stderr, "spi_write_then_read before spi_init\n");
		return -1;
	}
	/* The vendor always passes the two instruction bytes in txbuf, so a
	 * well-formed call has n_tx >= 2. */
	if (n_tx < 2 || n_tx > 2 + MAX_MBYTE_SPI || n_rx > MAX_MBYTE_SPI) {
		fprintf(stderr, "spi_write_then_read: bad lengths n_tx=%u n_rx=%u\n",
		        n_tx, n_rx);
		return -1;
	}
	n_data = n_tx - 2;

	instr = (uint16_t)(((uint16_t)txbuf[0] << 8) | txbuf[1]);

	for (i = 0; i < n_data && i < 4; i++)
		w0 |= (uint32_t)txbuf[2 + i] << (8 * i);
	for (i = 4; i < n_data; i++)
		w1 |= (uint32_t)txbuf[2 + i] << (8 * (i - 4));

	MMIO_WRITE(RF_SPI_A(RF_SPI_TXD0), w0);
	MMIO_WRITE(RF_SPI_A(RF_SPI_TXD1), w1);
	MMIO_WRITE(RF_SPI_A(RF_SPI_INSTR), instr);
	MMIO_WRITE(RF_SPI_A(RF_SPI_CTRL), (n_rx << 8) | n_data | (1u << 16));

	if (spi_wait_done() != 0)
		return -1;

	if (n_rx) {
		/* 勘误（2026-09-17 执行时修正）：MMIO_READ 返回 uint32_t，
		 * `<< 32` 是未定义行为（且实际会移位 0）。必须先提升到 64 位。 */
		full = (uint64_t)MMIO_READ(RF_SPI_A(RF_SPI_RXD0))
		     | ((uint64_t)MMIO_READ(RF_SPI_A(RF_SPI_RXD1)) << 32);
		for (i = 0; i < n_rx; i++)
			rx[i] = (uint8_t)((full >> (8 * i)) & 0xFFu);
		if (rxbuf) memcpy(rxbuf, rx, n_rx);
	}

	/* Trace the instruction word's own fields, not the raw bytes, so the log
	 * is a register-level history: W/R, address, data. */
	{
		int is_write = (instr >> 15) & 1;
		uint16_t addr = instr & 0x3FFu;
		if (is_write)
			trace_add(1, addr, &txbuf[2], n_data);
		else
			trace_add(0, addr, rx, n_rx);
	}
	return 0;
}

/* Single-byte read on the driver's old interface.  Unused by the vendor's own
 * code (ad9361.c goes through ad9361_spi_read -> spi_write_then_read), but it
 * is in platform.h, so it is implemented rather than left to link-fail. */
int32_t spi_read(struct spi_device *spi, uint8_t *data, uint8_t bytes_number)
{
	uint8_t dummy = 0;
	(void)dummy;
	/* The reference's version sends bytes_number bytes and reads them back;
	 * nothing in this driver reaches it, so it is a one-byte passthrough. */
	return spi_write_then_read(spi, data, bytes_number, data, bytes_number);
}

/* -------------------------------------------------------------------- GPIO
 *
 * Nine outputs on one axi_gpio_ctrl aperture, bit order per create_bd.tcl:
 *   bit0 txnrx  bit1 enable  bit2 resetb  bit3 sync  bit4 en_agc  bits8:5 ctl0-3
 * which is also the reference main.c's EMIO numbering, so pin number == bit.
 *
 * The reference platform.c's gpio_is_valid is `if (number >= 0) return 1;`,
 * which accepts -1 -- and the reference main.c does pass -1 for gpio_sync,
 * gpio_cal_sw1 and gpio_cal_sw2.  Tightened here to the nine pins that exist.
 */
static uint32_t gpio_shadow;      /* last value written, for RMW */
static int      gpio_ready;

void gpio_init(uint32_t device_id)
{
	(void)device_id;
	/* All nine are outputs; TRI is written all-zero once and never touched. */
	MMIO_WRITE(RF_GPIO_A(RF_GPIO_TRI), 0);
	gpio_shadow = 0;
	MMIO_WRITE(RF_GPIO_A(RF_GPIO_DATA), gpio_shadow);
	gpio_ready = 1;
}

void gpio_direction(uint8_t pin, uint8_t direction)
{
	/* Every pin is an output; the aperture has no per-bit direction.  Kept so
	 * a stray gpio_direction(pin, 0) in driver code is visible rather than
	 * silently ignored. */
	if (!gpio_is_valid(pin)) return;
	if (direction != 1)
		fprintf(stderr, "gpio_direction: pin %u asked for input, "
		                "but all nine mode pins are outputs\n", pin);
}

bool gpio_is_valid(int number)
{
	return (number >= 0 && number < RF_GPIO_COUNT) ? true : false;
}

void gpio_set_value(unsigned gpio, int value)
{
	if (!gpio_is_valid((int)gpio)) return;
	if (!gpio_ready) gpio_init(0);
	if (value) gpio_shadow |=  (1u << gpio);
	else       gpio_shadow &= ~(1u << gpio);
	MMIO_WRITE(RF_GPIO_A(RF_GPIO_DATA), gpio_shadow);
}
```

- [ ] **Step 6: 写 `selftest.c` —— Step 0 的判据，逐字照抄已 PASS 的 Python 三条**

`selftest.c` 是 spec §6.1 Step 0 的落地：**在任何厂商驱动代码被调用之前**，先证明我们的
C 传输层。判据**逐字照抄** `spi_probe.py` 已经 PASS 的 L1-a/L1-b 三条 —— 不是重新设计，
是复现。这样后面出问题就只剩「厂商驱动/寄存器序列」一个变量。

```c
/* selftest.c -- the C transport-layer self-check, run before any vendor code.
 *
 * This reproduces, byte for byte, the three criteria spi_probe.py already
 * passed on this board (docs/rf/boardtest.md L1-a/L1-b).  The point is not to
 * design a new test but to remove a variable: every later failure has two
 * possible sources -- the vendor driver's register sequence, and our C
 * transport layer.  Proving the latter separately leaves only one.
 *
 * Run on the board:
 *     sudo ./board_app selftest
 * Run on the host (same code, fake_pl instead of /dev/mem):
 *     make host-test
 */
#include "platform_rf.h"
#include "platform.h"
#include "ad9361.h"

#include <stdio.h>
#include <string.h>

/* ------------------------------------------------------------------ 判据表
 *
 * Written down here, before any measurement, per the project's report
 * discipline.  Each entry prints PASS/FAIL; nothing is inferred afterwards.
 *
 *   S1  PRODUCT_ID bits[7:3] == 0x08      (this board answers 0x0A = rev 2)
 *   S2  enable=1 released BEFORE resetb, and PRODUCT_ID is POLLED after the
 *       release because the part NACKs during its own self-check window
 *   S3  REG_CTRL (0x3DF) bit0 round-trips: write 0x01 -> read bit0 == 1,
 *       write 0x00 -> read bit0 == 0
 *
 * S1 is a MASK test, not equality: the vendor driver tests
 * (ret & 0xF8) != 0x08, where 0x08 is an in-place mask constant.  An equality
 * test against one specific byte would reject every other silicon revision.
 *
 * S3's target is REG_CTRL, not a scratch register: this part has no scratch
 * register.  0x3F5 is REG_OBSERVE_CONFIG, which the driver only ever writes
 * back what it just read -- using it as a scratch register produced a false
 * "the write path is broken" verdict on this board once already.
 */

/* 勘误（2026-09-17 执行时修正）：REG_PRODUCT_ID 与 REG_CTRL 在 ad9361.h 里
 * **已经定义过**（同值）。裸重定义会触发 -Wmacro-redefined 警告，所以加 guard。
 * 保留本地定义是为了让判据表不依赖厂商头文件的拼写。 */
#ifndef REG_PRODUCT_ID
#define REG_PRODUCT_ID   0x037u
#endif
#ifndef REG_CTRL
#define REG_CTRL         0x3DFu
#endif
#define AD9361_ID_MASK   0xF8u
#define AD9361_ID_9361   0x08u
#define AD9361_REV_MASK  0x07u

static int st_pass, st_fail;

static void st_chk(const char *name, int cond)
{
	if (cond) { st_pass++; printf("PASS %s\n", name); }
	else      { st_fail++; printf("FAIL %s\n", name); }
}

/* One-byte register access through the vendor's own instruction-word macros,
 * so the encoding under test is the vendor's, not a local re-derivation.
 *
 * 勘误（2026-09-17 执行时修正）：这两个 static 函数**不能**叫 rf_reg_read /
 * rf_reg_write —— 那正是 platform_rf.h 里 axi_ctrl_0 访问器的名字，会硬冲突
 * （同一翻译单元里重定义）。改成 st_ 前缀。 */
static uint8_t st_reg_read(struct spi_device *spi, uint32_t addr)
{
	uint8_t tx[2], rx[1];
	uint16_t cmd = AD_READ | AD_CNT(1) | AD_ADDR(addr);
	tx[0] = (uint8_t)(cmd >> 8);
	tx[1] = (uint8_t)(cmd & 0xFF);
	rx[0] = 0;
	spi_write_then_read(spi, tx, 2, rx, 1);
	return rx[0];
}

static int st_reg_write(struct spi_device *spi, uint32_t addr, uint8_t val)
{
	uint8_t tx[3];
	uint16_t cmd = AD_WRITE | AD_CNT(1) | AD_ADDR(addr);
	tx[0] = (uint8_t)(cmd >> 8);
	tx[1] = (uint8_t)(cmd & 0xFF);
	tx[2] = val;
	return spi_write_then_read(spi, tx, 3, NULL, 0);
}

/* The mode-pin sequence, in the order the reference main.c uses it:
 * gpio_initial() leaves txnrx=1, enable=0, resetb=1, sync=0, en_agc=0,
 * ctl0-3=0; then the driver drives resetb itself through ad9361_reset().
 * Here we drive it explicitly so the ordering is a criterion rather than a
 * side effect of the vendor driver. */
static void mode_pins_to_reset(void)
{
	gpio_set_value(RF_GPIO_TXNRX,   1);
	gpio_set_value(RF_GPIO_ENABLE,  1);   /* enable FIRST ... */
	gpio_set_value(RF_GPIO_RESETB,  0);   /* ... then assert reset ... */
	gpio_set_value(RF_GPIO_SYNC,    0);
	gpio_set_value(RF_GPIO_EN_AGC,  0);
	gpio_set_value(RF_GPIO_CTL0,    0);
	gpio_set_value(RF_GPIO_CTL1,    0);
	gpio_set_value(RF_GPIO_CTL2,    0);
	gpio_set_value(RF_GPIO_CTL3,    0);
}

int rf_selftest(void)
{
	struct spi_device spi;
	uint8_t id, rev, v;
	int i, seen = 0;

	st_pass = st_fail = 0;

	printf("=== RF transport self-check (spec 6.1 Step 0) ===\n");
	printf("axi_ctrl_0    @ 0x%08X\n", RF_AXI_CTRL_BASE);
	printf("pl_spi_0      @ 0x%08X\n", RF_PL_SPI_BASE);
	printf("axi_gpio_ctrl @ 0x%08X\n", RF_GPIO_CTRL_BASE);

	/* ---- transport bring-up.  gpio_init first: the part ignores SPI while
	 *      resetb is held, which looks exactly like a dead SPI master. ---- */
	memset(&spi, 0, sizeof(spi));
	spi.id_no = 0;

	gpio_init(0);
	mode_pins_to_reset();

	if (spi_init(0, 1, 0) != 0) {
		printf("FAIL spi_init -- cannot continue\n");
		return 1;
	}

	/* S2 ordering, asserted on the shadow value that was actually written:
	 * enable must already be high at the moment resetb goes high. */
	/* 勘误（2026-09-17 执行时修正）：原计划的表达式要求此刻 resetb **已为高**，
	 * 但 mode_pins_to_reset() 刚刚刻意把 resetb 拉低（复位已断言、释放是下一行），
	 * 所以那个表达式**在健康的模型上也必然失败** —— 它断言了一个此刻还不存在的状态，
	 * 自相矛盾。
	 *
	 * 改成"释放前那一刻能做出的最强陈述"：ENABLE 已为 1，且 RESETB 仍为 0。
	 * 这比原计划**更有区分力** —— 原计划那种事后快照，两种顺序都能满足；
	 * 而"释放前 enable 就已经是高"只有正确的顺序才满足。 */
	st_chk("S2a enable=1 is set before resetb is released",
	       (rf_gpio_shadow() & (1u << RF_GPIO_ENABLE)) != 0u &&
	       (rf_gpio_shadow() & (1u << RF_GPIO_RESETB)) == 0u);

	/* Release reset.  The part then runs its own self-check and NACKs reads
	 * for a short while -- poll, do not decide on the first attempt. */
	gpio_set_value(RF_GPIO_RESETB, 1);

	for (i = 0; i < 200; i++) {          /* up to 2 s at 10 ms */
		id = st_reg_read(&spi, REG_PRODUCT_ID);
		if ((id & AD9361_ID_MASK) == AD9361_ID_9361) { seen = 1; break; }
		msleep_interruptible(10);
	}

	st_chk("S2b PRODUCT_ID answered inside the polling window", seen);

	if (!seen) {
		printf("=== %d PASS, %d FAIL ===\n", st_pass, st_fail);
		printf("RESULT FAIL (device never answered -- no register was written)\n");
		return 1;
	}

	rev = id & AD9361_REV_MASK;
	printf("     PRODUCT_ID = 0x%02X (masked 0x%02X) rev = %u\n",
	       id, id & AD9361_ID_MASK, rev);

	/* S1: the mask test, exactly as the vendor driver does it. */
	st_chk("S1 PRODUCT_ID bits[7:3] == 0x08", (id & AD9361_ID_MASK) == AD9361_ID_9361);

	/* S3: the write path.  This is the criterion the CNT encoding bug broke,
	 * and it is the only one of the three that exercises a WRITE -- which is
	 * why a read-only probe looked perfectly healthy while every write died. */
	if (st_reg_write(&spi, REG_CTRL, 0x01) != 0) {
		printf("FAIL S3 write of REG_CTRL returned an error\n");
	} else {
		v = st_reg_read(&spi, REG_CTRL);
		st_chk("S3a write REG_CTRL 0x01 -> read bit0 == 1", (v & 0x1u) == 0x1u);

		st_reg_write(&spi, REG_CTRL, 0x00);
		v = st_reg_read(&spi, REG_CTRL);
		st_chk("S3b write REG_CTRL 0x00 -> read bit0 == 0", (v & 0x1u) == 0x0u);
	}

	printf("=== %d PASS, %d FAIL ===\n", st_pass, st_fail);
	printf("RESULT %s\n", st_fail ? "FAIL" : "PASS");
	return st_fail ? 1 : 0;
}
```

- [ ] **Step 7: 写 `test/test_selftest.c` —— 主机侧跑同一份 `rf_selftest()`**

```c
/* test_selftest.c -- run the board's own self-check on the host, against
 * fake_pl.c.  Same source file, same criteria; only the MMIO target differs.
 *
 * Three scenarios, because a self-check that only ever passes proves nothing:
 *   1. a healthy part must produce RESULT PASS
 *   2. a part that ignores writes (the CNT bug's signature) must produce FAIL
 *   3. a part that is not an AD9361 must fail S1 and write nothing
 * Scenarios 2 and 3 are what make scenario 1 meaningful. */
#include <stdio.h>
#include "fake_pl.h"
#include "platform_rf.h"

int rf_selftest(void);

static int fails;

int main(void)
{
	int rc;

	printf("=== test_selftest ===\n");

	/* ---- 1. healthy ---- */
	fake_pl_reset();
	fake_pl_set_product_id(0x0Au);        /* this board's rev 2 */
	rc = rf_selftest();
```

```c
/* Read the read-only STATUS word and wait for a fresh measurement window.
 *
 * The window is self-running in hardware (2^15 FCLK0 cycles = 655.36 us) and
 * publishes a rolling sequence number, so "is this reading fresh" is answered
 * by win_seq changing -- not by software timing, which cannot work here: one
 * mmap read costs ~25 us while DATA_CLK is ~33 ns, so software can only ever
 * observe "it is toggling", never count edges. */
static uint32_t status_read_fresh(void)
{
	uint32_t w0 = rf_reg_read(RF_REG_STATUS);
	uint32_t seq0 = RF_STATUS_WIN_SEQ(w0);
	int i;

	for (i = 0; i < 2000; i++) {
		uint32_t w = rf_reg_read(RF_REG_STATUS);
		if (RF_STATUS_WIN_SEQ(w) != seq0 && RF_STATUS_WIN_DONE(w))
			return w;
		udelay(100);                    /* 2000 * 100us = 200 ms max */
	}
	/* Never advanced: report the stale word so the caller can say so. */
	return w0;
}
```

```c
	        "usage: board_app <subcommand>\n"
	        "  selftest               transport self-check (no vendor code)\n"
	        "  status                 read the PL STATUS word\n"
	        "  plcfg                  spec 6.1 Step 1: PL config + NEGATIVE check\n"
	        "  init [trace.txt]       spec 6.1 Step 2: ad9361_init + parameters\n"
	        "  atten <mdB>            set TX attenuation on both channels\n"
	        "  lo <Hz>                set TX LO frequency\n"
	        "  phase_inc <n>          set the PL DDS phase increment (12 bit)\n"

	if (!strcmp(argv[1], "selftest"))
```

- [ ] **Step 1: 写脚本骨架与 B 层判据**

```python
#!/usr/bin/env python3
"""AD9361 TX bring-up: the B/C-layer scope criteria (spec 6.1 Step 4).

Run this with the SYSTEM python -- it imports pyvisa:

    python -u rf_overlay/sw/scope_tx_check.py --board-app /home/xilinx/board_app_src/board_app

The criteria below are copied from the design spec (docs/superpowers/specs/
2026-09-17-ad9361-tx-bringup-design.md section 6.1 Step 4) and were fixed
BEFORE any measurement.  If a criterion turns out to be badly chosen, the fix
is to record it as NOT RUN and revisit, never to move the threshold after
seeing the number.

Everything is judged RELATIVE -- position, difference, ratio -- never absolute
amplitude.  The 80 MHz carrier sits in the roll-off of this scope's 100 MHz
front end, whose response is not characterised here, so an absolute amplitude
threshold would be a criterion written against an unknown.

Layer B is DIAGNOSTIC and frequency-only.  Its sole signal source is LO
leakage: there is no chip-internal tone generator in this part (spec 0.2 --
ad9361.c/.h have zero DDS references), and the leakage amplitude was never
measured, so no amplitude threshold can be justified.  "Cannot resolve a line
at LO" means "instrument-limited, this diagnostic is unavailable", which is
NOT a failure.  B's value is bisection: if C fails while B passes, the analog
chain is alive and the problem is in the data path.

Layer C is the primary criterion.
"""

import argparse
import sys
import time

import numpy as np

# ---------------------------------------------------------------- criteria
# Frozen before measurement.  Kept as named constants so that a change is a
# visible diff rather than a tweak buried in a comparison.
WIN_LOG2          = 15          # from the PL STATUS register's WIN_LOG2 field
FCLK0_HZ          = 50_000_000  # FCLK0, read from SLCR per docs/rf/boardtest.md

LO_NOMINAL_HZ     = 80e6
LO_SHIFTED_HZ     = 85e6

# 4.2: phase_inc 128 -> 256 must double the tone offset.
PHASE_INC_1       = 128
PHASE_INC_2       = 256
DF_RATIO_MIN      = 1.96        # 2.00 - 2%
DF_RATIO_MAX      = 2.04        # 2.00 + 2%

# 4.5: atten 10000 -> 30000 mdB is a 20 dB change; take the DIFFERENCE so the
# scope's front-end response cancels.
ATTEN_LOW_MDB     = 10000
ATTEN_HIGH_MDB    = 30000
ATTEN_DELTA_DB    = 20.0
ATTEN_TOL_DB      = 3.0

# B2/4.4: the LO moves 80 -> 85 MHz; every line must move with it.
LO_SHIFT_TOL_HZ   = 2e6

# Peak-picking: a line must stand this far above the local noise floor.
PEAK_SNR_MIN_DB   = 6.0

# Timebase.  :WAV:POIN must NEVER be written (it pins the record length to a
# manual value that :WAV:POIN:MODE NORM cannot reset, after which every
# waveform-capture criterion in every suite silently misjudges).  A short
# timebase is what guarantees a high sample rate; 2 ms/div would push the
# sample rate to a few MSa/s and 80 MHz would vanish entirely.
#
# 6.25 µs/div, not 20 ns/div (plan defect E18).  The spec's constraint is
# 62500 points in ~62 µs at >= 500 MSa/s, i.e. a 16 kHz bin -- and it is the
# BIN WIDTH that decides whether the sidebands are resolvable.  At 20 ns/div
# the window is 200 ns, and 62500 points over 200 ns would need 312.5 GSa/s;
# the DSOX1102A caps at 2 GSa/s, so it silently shortens the record instead
# and the bin becomes ~5 MHz wide.  Every candidate sideband (0.47 / 0.94 /
# 1.88 MHz) is then well under one bin, and C1/C2/C3 would report "no lines"
# on a perfectly good transmitter -- the failure would look like broken RTL.
TIM_SCAL          = 6.25e-6    # 6.25 us/div, 10 div -> 62.5 us window @ 1 GSa/s
CHAN_SCAL         = 0.5        # relative only; see the module docstring


def peak_freqs(freqs, spec, n=8, min_snr_db=PEAK_SNR_MIN_DB):
    """The n strongest local maxima, each above the median floor by min_snr_db.

    Returned as a list of (freq_hz, magnitude, snr_db) sorted by magnitude.
    The median is used as the floor because a spectrum with a few strong lines
    has a median dominated by the noise between them.
    """
    floor = np.median(spec)
    if floor <= 0:
        floor = 1e-12
    cand = []
    for k in range(1, len(spec) - 1):
        if spec[k] > spec[k - 1] and spec[k] >= spec[k + 1]:
            snr = 20.0 * np.log10(max(spec[k], 1e-12) / floor)
            if snr >= min_snr_db:
                cand.append((freqs[k], spec[k], snr))
    cand.sort(key=lambda t: -t[1])
    return cand[:n]


def find_nearest(peaks, target_hz, tol_hz):
    """The strongest peak within tol of target, or None."""
    best = None
    for f, m, snr in peaks:
        if abs(f - target_hz) <= tol_hz:
            if best is None or m > best[1]:
                best = (f, m, snr)
    return best
```

```python
class Report(object):
    def __init__(self):
        self.rows = []

    def add(self, name, verdict, detail=""):
        self.rows.append((name, verdict, detail))
        print("%-6s %-46s %s" % (verdict, name, detail))

    def count(self, verdict):
        return sum(1 for r in self.rows if r[1] == verdict)

    def summary(self):
        p, f, n = self.count("PASS"), self.count("FAIL"), self.count("N/A")
        print("\n=== %d PASS, %d FAIL, %d NOT-APPLICABLE ===" % (p, f, n))
        if n:
            print("NOT-APPLICABLE means instrument-limited, not a failure "
                  "(spec 6.1 B layer).")
        return f


def layer_b(rep, wav_fft, lo_hz, label):
    """B1/B2: a resolvable line at LO, and it moves when LO moves.

    Frequency only.  No amplitude threshold -- see the module docstring.
    """
    freqs, spec = wav_fft("CHAN1")
    peaks = peak_freqs(freqs, spec)
    hit = find_nearest(peaks, lo_hz, tol_hz=LO_SHIFT_TOL_HZ)

    if hit is None:
        rep.add("B1 (%s) line resolvable at LO" % label, "N/A",
                "strongest lines: %s" % ", ".join("%.1f MHz" % (f / 1e6)
                                                  for f, _, _ in peaks[:4]))
        return None
    rep.add("B1 (%s) line resolvable at LO" % label, "PASS",
            "%.3f MHz (target %.1f MHz, snr %.1f dB)" % (hit[0] / 1e6,
                                                         lo_hz / 1e6, hit[2]))
    return hit[0]


def layer_c(rep, scope, board, wav_fft):
    """C1..C6 -- the primary criteria.  See spec 6.1 Step 4's table."""
    import board_cmd

    # --- measurement 1: baseline, LO = 80 MHz, atten 0, phase_inc 128 ---
    # atten 0 rather than 10000: LO leakage is all layer B has, and 10 dB
    # spent up front is 10 dB of visibility thrown away.
    board.cmd("atten 0")
    board.cmd("lo %d" % int(LO_NOMINAL_HZ))
    board.cmd("phase_inc %d" % PHASE_INC_1)
    time.sleep(0.3)

    freqs, spec = wav_fft("CHAN1")
    peaks = peak_freqs(freqs, spec)
    lo_line = find_nearest(peaks, LO_NOMINAL_HZ, LO_SHIFT_TOL_HZ)

    # C1: presence.  LO leakage plus a symmetric pair at LO +/- df.  Two
    # sidebands, not one: the DDS is I-only, so dac_data_q1 is hard zero and
    # the output is double-sideband (spec 7).  A single sideband would mean
    # something other than this DDS is producing the tone.
    side = [p for p in peaks
            if lo_line is not None and abs(p[0] - lo_line[0]) > LO_SHIFT_TOL_HZ]
    lower = [p for p in side if p[0] < (lo_line[0] if lo_line else LO_NOMINAL_HZ)]
    upper = [p for p in side if p[0] > (lo_line[0] if lo_line else LO_NOMINAL_HZ)]

    if lo_line and lower and upper:
        # pick the closest symmetric pair
        best = None
        for l in lower:
            for u in upper:
                if lo_line is None:
                    continue
                sym = abs((lo_line[0] - l[0]) - (u[0] - lo_line[0]))
                if best is None or sym < best[0]:
                    best = (sym, l, u)
        if best is not None:
            _, l, u = best
            df1 = lo_line[0] - l[0]
            rep.add("C1 spectrum lines present (carrier + symmetric pair)", "PASS",
                    "LO %.3f MHz, sidebands %.3f / %.3f MHz, df = %.1f kHz"
                    % (lo_line[0] / 1e6, l[0] / 1e6, u[0] / 1e6, df1 / 1e3))
        else:
            rep.add("C1 spectrum lines present", "FAIL", "no symmetric pair")
            df1 = None
    else:
        rep.add("C1 spectrum lines present", "FAIL",
                "peaks: %s" % ", ".join("%.1f MHz" % (f / 1e6) for f, _, _ in peaks[:6]))
        df1 = None

    # C2: df must be LINEAR in phase_inc.  This is what separates "a DDS tone"
    # from "a spurious line that happens to sit near LO".
    df2 = None
    if df1 is not None and df1 > 0:
        board.cmd("phase_inc %d" % PHASE_INC_2)
        time.sleep(0.3)
        freqs2, spec2 = wav_fft("CHAN1")
        peaks2 = peak_freqs(freqs2, spec2)
        lo2 = find_nearest(peaks2, LO_NOMINAL_HZ, LO_SHIFT_TOL_HZ)
        if lo2 is not None:
            low2 = [p for p in peaks2 if p[0] < lo2[0] - LO_SHIFT_TOL_HZ]
            if low2:
                df2 = lo2[0] - max(low2, key=lambda p: p[1])[0]
        if df2 is not None and df2 > 0:
            ratio = df2 / df1
            ok = DF_RATIO_MIN <= ratio <= DF_RATIO_MAX
            rep.add("C2 df is linear in phase_inc (ratio 2.00 +/- 2%)",
                    "PASS" if ok else "FAIL",
                    "df1 = %.1f kHz, df2 = %.1f kHz, ratio = %.4f"
                    % (df1 / 1e3, df2 / 1e3, ratio))
        else:
            rep.add("C2 df is linear in phase_inc", "FAIL", "no sideband at 256")
    else:
        rep.add("C2 df is linear in phase_inc", "FAIL", "C1 gave no df1")

    # C3: which candidate does df actually match?  The two comments in our own
    # RTL disagree by a factor of two (spec 0.1), so this is MEASURED, not
    # assumed.  Record which one it is either way.
    #
    # THE FIR IS ENABLED (per the example, per the user's direction), so the bare
    # df = inc/2^12 * fs relation is not directly trustworthy: the interpolating
    # TX FIR has its own passband, and a tone near its edge is attenuated and
    # slightly phase-shifted.  Two things protect this criterion:
    #
    #   1. C2 already established that df scales EXACTLY with phase_inc (ratio
    #      2.00 +/- 2%).  That is a property of the DDS, and the FIR -- being a
    #      fixed LTI filter -- cannot produce it.  So the line IS the DDS tone.
    #   2. Both candidate tones are placed well inside the passband, so the FIR's
    #      effect on their POSITION is nil; only their amplitudes differ.  Peak
    #      position is what C3 uses.
    #
    # What the FIR does break is any attempt to read a candidate's amplitude as a
    # proxy for its frequency.  C3 therefore never compares amplitudes between the
    # two candidates; it compares the measured df against the two arithmetic
    # candidates and reports the closer one, with the margin printed so a
    # near-tie is visible rather than hidden.
    #
    # ⚠ 计划缺陷 E11 —— 下面的两组候选**不完整，且会绿着印出错误结论**，
    #   照抄会重犯。`docs/rf/example_project_notes.md:85` 记载 LVDS DDR 下
    #   60 MSPS 的 DATA_CLK 是 30 MHz，即 **fs = 2 x f_DATA_CLK**；这一组不在
    #   候选里。若真值就是 2，则 df1 = 2*cand_a，于是 da=ca、db=1.5ca、da<db，
    #   代码会印 "A (fs == f_DATA_CLK)" 而 margin = 1.0 —— **判 PASS 且结论是错的**。
    #   正确做法见 `.superpowers/sdd/2026-09-17-ad9361-tx-bringup/
    #   dispatch-notes-task5.md` 第 7 节：枚举 {1/2, 1, 2} 三个比值、取最近者、
    #   仍要求与次近者清晰分离、**并把识别出的比值以数字印出**；文档期望值只作
    #   RECORD 交叉核对，不作判据（该文档自己就说来源约束可能是保守值）。
    if df1 is not None and df1 > 0:
        # candidate A: fs == f_DATA_CLK        -> df = inc/2^12 * f_DATA_CLK
        # candidate B: fs == f_DATA_CLK / 2    -> df = inc/2^12 * f_DATA_CLK / 2
        f_dclk = read_f_data_clk_hz(board)
        if f_dclk:
            cand_a = PHASE_INC_1 / float(1 << 12) * f_dclk
            cand_b = cand_a / 2.0
            da, db = abs(df1 - cand_a), abs(df1 - cand_b)
            which = "A (fs == f_DATA_CLK)" if da < db else "B (fs == f_DATA_CLK/2)"
            # A tie or a near-tie means the measurement cannot decide, and saying
            # so is more useful than picking one.  10% of the smaller candidate
            # separates them: the two candidates differ by 2x, so anything under
            # 10% is far closer than the 100% gap that distinguishes them.
            margin = abs(da - db) / max(min(cand_a, cand_b), 1.0)
            verdict = "PASS" if margin > 0.10 else "FAIL"
            rep.add("C3 fs <-> DATA_CLK relation adjudicated by measurement",
                    verdict,
                    "df = %.1f kHz; f_DATA_CLK = %.3f MHz; candidate A = %.1f kHz, "
                    "candidate B = %.1f kHz -> matches %s (separation %.1f%% of the "
                    "smaller candidate%s)"
                    % (df1 / 1e3, f_dclk / 1e6, cand_a / 1e3, cand_b / 1e3, which,
                       margin * 100.0,
                       "" if verdict == "PASS" else " -- TOO CLOSE TO CALL"))
        else:
            rep.add("C3 fs <-> DATA_CLK relation", "FAIL",
                    "PL STATUS reported no DATA_CLK; run `board_app status`")
    else:
        rep.add("C3 fs <-> DATA_CLK relation", "FAIL", "no df to adjudicate")

    # C4: move LO and require EVERY line to move with it.  This is what shows
    # the lines are baseband content up-converted, not artifacts of the scope
    # or of the LO itself.
    if lo_line is not None:
        before = [p[0] for p in peaks[:6]]
        board.cmd("lo %d" % int(LO_SHIFTED_HZ))
        time.sleep(0.3)
        freqs3, spec3 = wav_fft("CHAN1")
        peaks3 = peak_freqs(freqs3, spec3)
        shifted_ok = True
        details = []
        for f0 in before:
            target = f0 + (LO_SHIFTED_HZ - LO_NOMINAL_HZ)
            hit = find_nearest(peaks3, target, tol_hz=LO_SHIFT_TOL_HZ)
            details.append("%.1f->%s" % (f0 / 1e6,
                                         "%.1f" % (hit[0] / 1e6) if hit else "MISSING"))
            if hit is None:
                shifted_ok = False
        rep.add("C4 all lines translate with LO (+5 MHz)", "PASS" if shifted_ok else "FAIL",
                " ".join(details))
        board.cmd("lo %d" % int(LO_NOMINAL_HZ))
        time.sleep(0.3)
    else:
        rep.add("C4 all lines translate with LO", "FAIL", "no baseline lines")

    # C5: attenuation difference.  A DIFFERENCE, so the scope's front-end
    # response cancels -- the whole point of judging relatively.
    if lo_line is not None:
        board.cmd("atten %d" % ATTEN_LOW_MDB)
        time.sleep(0.3)
        _, s_low = wav_fft("CHAN1")
        board.cmd("atten %d" % ATTEN_HIGH_MDB)
        time.sleep(0.3)
        _, s_high = wav_fft("CHAN1")

        # Compare the same spectral bin in both captures: take the peak bin
        # from the low-attenuation capture and read it in both.
        k = int(np.argmax(s_low))
        a_low  = max(s_low[k], 1e-12)
        a_high = max(s_high[k], 1e-12)
        delta = 20.0 * np.log10(a_low / a_high)
        ok = abs(delta - ATTEN_DELTA_DB) <= ATTEN_TOL_DB
        rep.add("C5 atten 10000->30000 gives a 20 +/- 3 dB difference",
                "PASS" if ok else "FAIL",
                "measured difference = %.2f dB (bin %.3f MHz)"
                % (delta, freqs[k] / 1e6))

        # C6: LO leakage vs the signal lines.  RECORD ONLY, never judge: it
        # says where the attenuator sits relative to the leakage path, which
        # is informative but not something this round set a criterion for.
        lo_peak_low = find_nearest(peak_freqs(freqs, s_low), LO_NOMINAL_HZ, LO_SHIFT_TOL_HZ)
        lo_peak_high = find_nearest(peak_freqs(freqs, s_high), LO_NOMINAL_HZ, LO_SHIFT_TOL_HZ)
        if lo_peak_low and lo_peak_high:
            rep.add("C6 LO leakage vs attenuation", "RECORD",
                    "%.2f dB change (NOT a criterion -- spec 6.1 C6)"
                    % (20.0 * np.log10(lo_peak_low[1] / max(lo_peak_high[1], 1e-12))))
        else:
            rep.add("C6 LO leakage vs attenuation", "RECORD",
                    "leakage line not resolvable in one or both captures")
    else:
        rep.add("C5 atten difference", "FAIL", "no baseline lines")
        rep.add("C6 LO leakage vs attenuation", "N/A", "no baseline lines")


def read_f_data_clk_hz(board):
    """f_DATA_CLK from the PL STATUS register, via `board_app status`."""
    out = board.cmd("status")
    for line in out.splitlines():
        if "f_DATA_CLK" in line and "NO CLOCK" in line:
            return 0
        if "f_DATA_CLK" in line and "Hz" in line:
            try:
                return float(line.split("=")[1].split("Hz")[0].strip())
            except (IndexError, ValueError):
                return 0
    return 0
```

```python
def capture_fft(inst, channel):
    """Freeze, read the NORM waveform, and return (freqs, |spectrum|).

    :WAV:POIN is deliberately never written here.  Writing it pins the record
    length to a manual value and :WAV:POIN:MODE NORM does not reset it, after
    which every waveform-based criterion in every suite misjudges at once --
    this project has already lost a session to that.
    """
    inst.write(":STOP")
    inst.write(":WAV:SOUR " + channel)
    time.sleep(0.05)
    xinc = float(inst.query(":WAV:XINC?"))
    data = inst.query_binary_values(":WAV:DATA?", datatype="B", container=list)
    inst.write(":RUN")
    y = np.asarray(data, dtype=float)
    y = y - y.mean()
    n = len(y)
    spec = np.abs(np.fft.rfft(y * np.hanning(n)))
    return np.fft.rfftfreq(n, xinc), spec


def scope_prepare(inst):
    """Fixed timebase and vertical scale.  Auto-scale converges too slowly
    after a signal change and can capture an incomplete period."""
    inst.write(":TIM:SCAL %.9E" % TIM_SCAL)
    inst.write(":CHAN1:SCAL %.9E" % CHAN_SCAL)
    inst.write(":CHAN1:OFFS 0")
    inst.write(":CHAN1:PROB 1")
    inst.write(":CHAN1:DISP ON")
    inst.write(":CHAN2:SCAL %.9E" % CHAN_SCAL)
    inst.write(":CHAN2:OFFS 0")
    inst.write(":CHAN2:PROB 1")
    inst.write(":RUN")
    time.sleep(0.5)
    # Verify the sample rate actually reached the >= 500 MSa/s the FFT needs.

    print("[scope] sample rate %.3f GSa/s (%s)" % (srate / 1e9, "OK" if srate >= 5e8
```

## Task 6: `audit_report.py` —— Step 5 的事后回读审计

**Files:**
- Create: `rf_overlay/sw/audit_report.py`

**Interfaces:**
- Consumes: `trace_init.txt`（Task 4 的 `init` 写出）与一次独立的回读遍历
- Produces: 三分类报告 —— **已证实 / 已声明豁免 / 无法解释的不符**

**为什么审计不能插在每次写后面**（spec §5.3 / R6）：VCO 校准有**时间窗**，
在每次写后面插一次回读会把 SPI 流量翻倍塞进状态机内部，可能让校准错过时间窗。
所以分成两趟：trace 常开（零时序改变，只记内存），audit 事后对**一份预先圈定的地址清单**
重读。

**纪律**：audit **不自动修正、不因失败中止**。无法解释的不符就是它最有价值的输出。

- [ ] **Step 1: 写 `audit_report.py`**

```python
#!/usr/bin/env python3
"""Post-init register read-back audit (spec 6.1 Step 5, spec 5.3).

The trace and the audit are two separate passes on purpose.  A read-back after
every write would double the SPI traffic inside the VCO calibration state
machine, and that calibration has a timing window -- inserting reads there
perturbs exactly what this round is trying to measure.  So the always-on trace
records transactions in memory (zero timing change) and this script re-reads a
PRE-ENUMERATED address list afterwards.

The exemption list below was fixed before the measurement, not derived from
what the read-back happened to show.  Deriving it afterwards would make the
audit a tautology.

Discipline: this script does not correct anything and does not abort on a
mismatch.  An unexplained mismatch is the most valuable thing it can report.

    python -u rf_overlay/sw/audit_report.py --trace trace_init.txt \
        --board-app /home/xilinx/board_app_src/board_app --board <host>
"""

import argparse
import re
import sys

# ---------------------------------------------------------------- exemptions
# Every entry needs a REASON.  "It did not match" is not a reason.
#
# (addr, mask, kind, reason)
#   kind "volatile" -- the register legitimately changes on its own
#   kind "weak"     -- the written value is 0 inside the mask, so "reads back 0"
#                      cannot prove the write took effect (this project has a
#                      memory entry on exactly this blind spot)
EXEMPT = [
    (0x037, 0xFF, "volatile",
     "PRODUCT_ID is read-only silicon revision, never written"),
    (0x3F4, 0xFF, "volatile",
     "BIST control / status; the part updates it"),
    (0x3F5, 0xFF, "volatile",
     "REG_OBSERVE_CONFIG -- the driver only writes back what it read"),
    (0x3F6, 0xFF, "volatile",
     "BIST-related status"),
    (0x00A, 0xFF, "volatile",
     "temperature / calibration status updates on its own"),
    (0x03E, 0xFF, "weak",
     "REG_LVDS_INVERT_CTRL2 -- written 0, so a 0 read-back is not evidence"),
]

# The registers the audit re-reads.  Enumerated from the trace's write history,
# which is exactly why the trace exists: it is the list of what we claim to
# have written.
def addresses_written(trace_path):
    """Parse the trace into {addr: last_value_written} plus a read count."""
    last = {}
    order = []
    with open(trace_path) as f:
        for line in f:
            m = re.match(r"^([WR]) ([0-9A-F]{3})((?: [0-9A-F]{2})*)$", line.strip())
            if not m:
                continue
            rw, addr, data = m.group(1), int(m.group(2), 16), m.group(3).split()
            if rw == "W":
                if addr not in last:
                    order.append(addr)
                # multi-byte writes take the LAST byte, matching the part
                last[addr] = int(data[-1], 16) if data else 0
    return last, order


def classify(addr, written, read_back):
    """Return (verdict, detail)."""
    for a, mask, kind, reason in EXEMPT:
        if a == addr:
            if kind == "weak":
                return "DECLARED-WEAK", "%s: %s" % (reason, "read %02X" % read_back)
            return "DECLARED-VOLATILE", "%s: %s" % (reason, "read %02X" % read_back)
    if (read_back & 0xFF) == (written & 0xFF):
        return "CONFIRMED", "wrote %02X, read %02X" % (written, read_back)
    return "UNEXPLAINED", "wrote %02X, read %02X" % (written, read_back)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--trace", required=True)
    ap.add_argument("--board-app", default="/home/xilinx/board_app_src/board_app")
    ap.add_argument("--board", default=None,
                    help="omit for an offline classification of the trace alone")
    ap.add_argument("--out", default="audit_report.txt")
    args = ap.parse_args()

    last, order = addresses_written(args.trace)
    print("trace: %d distinct registers written" % len(last))

    board = None
    if args.board:
        from board_cmd import BoardCmd
        board = BoardCmd(args.board_app, args.board)
        # The audit reads through the SAME driver instance the init used, so
        # run it as a subcommand that keeps the phy alive.  `dump` reads the
        # driver's cached view; a raw re-read needs `board_app audit <addr>`.
        # If the board is not given, this script classifies the trace alone and
        # says so rather than pretending it read anything.
        board.cmd("audit-begin")

    confirmed = weak = volatile = unexplained = 0
    lines = []
    for addr in order:
        written = last[addr]
        if board is None:
            verdict, detail = "NOT-READ", "wrote %02X; no board given" % written
        else:
            raw = board.cmd("audit %03X" % addr)
            try:
                read_back = int(raw.strip().split()[-1], 16)
            except (IndexError, ValueError):
                verdict, detail = "READ-FAILED", "raw output: %r" % raw[:80]
                read_back = None
            else:
                verdict, detail = classify(addr, written, read_back)

        if   verdict == "CONFIRMED":          confirmed += 1
        elif verdict == "DECLARED-WEAK":      weak += 1
        elif verdict == "DECLARED-VOLATILE":  volatile += 1
        elif verdict == "UNEXPLAINED":        unexplained += 1

        line = "%-18s 0x%03X  %s" % (verdict, addr, detail)
        lines.append(line)
        print(line)

    if board is not None:
        board.cmd("audit-end")

    summary = (
        "\n=== audit summary ===\n"
        "  confirmed          : %d\n"
        "  declared volatile  : %d\n"
        "  declared weak      : %d\n"
        "  UNEXPLAINED        : %d\n"
        % (confirmed, volatile, weak, unexplained))
    print(summary)

    with open(args.out, "w") as f:
        f.write("\n".join(lines) + "\n" + summary)

    if unexplained:
        print("Unexplained mismatches are NOT failures of this run -- they are "
```

- Produces: `phy_mode` 一路同时驱动 `ad9361_phy_0` 与 `rf_tx_source_0`；

**前置事实：在 1R1T 下 TX2 通道是关闭的，CH2 上预期什么都看不到。**
而"看不到"与"通道坏了"长得一模一样 ⇒ **当前配置下 CH2 有无信号都不构成判据**。
要让 TX2 有判断力，AD9361 必须进 2R2T。

**为什么这一次重建必须和阶段一分开、且只重建一次**（Global Constraints 第 2 条）：
`dds_ce` 周期不跟随 `phy_mode` 时，**TX1 的音会频率翻倍** —— 症状是"C 层判据突然差 2 倍"，
极易被当成器件问题。所以 `dds_ce` 的改动必须和 2R2T 启用**在同一次重建里**完成；
否则中间那次重建的 TX1 音频率是错的。

- [ ] **Step 1: 写失败的 TB 用例（`dds_ce` 周期跟随 `phy_mode`）**

在 `rf_overlay/tb/rf_tx_source_tb.v` 里加一组用例（若该 TB 不存在则新建，
只需例化 `rf_tx_source` 并观察 `dac_valid` 的占空比）：

```verilog
	// ---- dds_ce period must follow phy_mode ----
	//
	// In 1R1T the phy frames one I/Q sample every 2 DATA_CLK cycles; in 2R2T it
	// takes 4 (phy_tx.v:68, tx_data_cnt_reg wraps at 'd3).  The DDS must advance
	// once per SAMPLE, so its strobe period has to match.  Getting this wrong
	// doubles TX1's tone frequency in 2R2T, which reads as "the C-layer criterion
	// suddenly disagrees by 2x" -- easy to blame on the part instead of the RTL.
	//
	// Measured as: count DATA_CLK cycles between two consecutive dac_valid
	// rising edges (dac_valid is held high for the whole sample period).
	begin
		phy_mode = 1'b1;                     // 1R1T -> 2 cycles per sample
		@(negedge rst_n); @(negedge rst_n);
		n_cycles = 0;
		@(posedge tx_clk);                   // align
		wait (dac_valid === 1'b1);
		while (dac_valid === 1'b1) begin @(posedge tx_clk); n_cycles = n_cycles + 1; end
		chk("1R1T: sample period is 2 DATA_CLK cycles", n_cycles == 2);

		phy_mode = 1'b0;                     // 2R2T -> 4 cycles per sample
		@(negedge rst_n); @(negedge rst_n);
		n_cycles = 0;
		@(posedge tx_clk);
		wait (dac_valid === 1'b1);
		while (dac_valid === 1'b1) begin @(posedge tx_clk); n_cycles = n_cycles + 1; end
		chk("2R2T: sample period is 4 DATA_CLK cycles", n_cycles == 4);
	end
```

```verilog
	// The phy frames one I/Q sample every 2 DATA_CLK cycles in 1R1T and every 4
	// in 2R2T (phy_tx.v: tx_data_cnt_reg wraps at 'd1 vs 'd3).  The DDS must
	// advance once per SAMPLE, so the strobe period follows phy_mode.  Without
	// this, switching to 2R2T doubles TX1's tone frequency -- a symptom that
	// looks like a device fault rather than an RTL one.
	wire [7:0] strobe_max = phy_mode ? 8'd1 : 8'd3;

	reg        dds_ce;
	reg [7:0]  strobe_cnt;
	always @(posedge tx_clk or negedge rst_n) begin
		if (!rst_n) begin
			dds_ce     <= 1'b0;
			strobe_cnt <= 8'd0;
		end else begin
			dds_ce     <= (strobe_cnt == 8'd0);
			strobe_cnt <= (strobe_cnt == strobe_max) ? 8'd0 : strobe_cnt + 8'd1;
		end
	end
```

> `dds_ce` 为高，且 `dac_valid` 在整个样点周期里保持高。改上限**不改这个关系** ——
> 只改周期长度。执行者改完后**用 Step 1 的 TB 确认 1R1T 下周期仍是 2 拍**（回归），
> 这一条比新增的 2R2T 用例更重要：它是阶段一已经测过的配置。

- [ ] **Step 8: 给 `board_app` 加 `2r2t` 子命令**

```c
	/* ---- 2r2t: stage two.  Switch the AD9361 and the PL to 2R2T so TX2 has
	 *      a data path at all -- in 1R1T TX2 is off, and "no signal on CH2"
	 *      then means nothing, because a healthy off channel and a broken one
	 *      look identical.
	 *
	 *      ORDER MATTERS: phy_mode is written while the TX data path is quiet.
	 *      It reaches rf_tx_source's strobe divider combinationally, so a
	 *      mid-stream change would glitch one sample period. ---- */
	if (!strcmp(argv[1], "2r2t")) {
		/* Stop the tone first: phase_inc = 0 makes the DDS emit DC, so the
		 * data path carries no edges while phy_mode changes. */
		rf_reg_write(RF_REG_PHASE_INC, 0);
		udelay(100);

		rf_reg_write(RF_REG_PHY_MODE, 0);      /* PL side: 2R2T framing */

		if (phy) {
			/* AD9361 side.  two_rx_two_tx_mode_enable is a struct field,
			 * so the API's own path is ad9361_set_trx_path_clks plus a
			 * re-init; the supported route is to re-run init with the flag
			 * set, which is what this does. */
			rf_init_param.two_rx_two_tx_mode_enable = 1;
			printf("re-initialising with two_rx_two_tx_mode_enable = 1\n");
			if (ad9361_init(&phy, &rf_init_param) != 0) {
				printf("RESULT FAIL: 2R2T re-init failed\n");
				return 1;
			}
			/* Same FIR configuration as the 1R1T path, from the same
			 * tables -- stage two must differ from stage one ONLY in
			 * phy_mode, or a CH2 difference has two explanations. */
			ad9361_set_tx_fir_config(phy, rf_tx_fir_config);
			ad9361_set_rx_fir_config(phy, rf_rx_fir_config_search);
			ad9361_set_tx_fir_en_dis(phy, 1);
			ad9361_set_rx_fir_en_dis(phy, 1);
		}

		printf("phy_mode readback = %u (expect 0)\n", rf_reg_read(RF_REG_PHY_MODE));
		printf("RESULT PASS if readback is 0 -- then set atten 0 and restore "
		       "phase_inc before measuring\n");
		return 0;
	}
```

**E4 与 E6 是同一类**：判据（TB / 模型）与实现的语义不一致，而症状是"测试通过"。
两次都是靠**执行者不肯将就**才发现的 —— 计划里那句"不要为了让测试变绿去放宽 mask 或阈值"
在这两处救了场。

**E6 的教训值得单独记**：`fake_pl.c` 建模的是**器件侧**，而器件的 CNT 规则只有
"一次写必须恰好带 `CNT+1` 个数据字节"这一条。**模型里任何多余的宽容或多余的严格都会
让反向用例失去意义。** 验证方法是**同时**看正向与反向用例的 PASS 行：
正向必须 `rejected=0`，反向必须 `rejected=1`。只有反向绿、正向也绿才算真绿。

**E7/E9 是第三种形态，比 E1–E6 更值得记**：它们不在被测对象里，也不在判据里，
而在**控制者自己的扫描过程**里。我在派发任何实现之前跑过一次冲突扫描并写了逐行结论，
但其中两行的结论是在**只读了一侧**的情况下写下的 —— 一行只看了调用点，一行的前提
（`session` 可行）从未验证。**扫描表的价值全在于"这一行两侧都读过了"，
写成"看起来没问题"就比不扫更坏，因为它会让人不再去看。**
判别方法很简单：任何"某任务需要的东西没人提供"式的结论，必须先能指出**被调用方在哪里**；
指不出来就不是结论，是待查项。

E9 另外说明了一件事：**"没有两个任务写同一文件"这类全局断言，必须在每个任务的
`Files:` 段都生成完之后再核**，而不是按任务顺序边读边下结论 —— 后面任务的改动
会推翻前面已经写下的全局断言。
