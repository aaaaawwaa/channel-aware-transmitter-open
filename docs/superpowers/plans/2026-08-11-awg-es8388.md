- [ ] **Step 1: 写生成脚本**

`tools/gen_sine.py`(完整内容):
```python
#!/usr/bin/env python3
"""Generate a 256 x 16-bit signed sine ROM as Verilog (src/rtl/sine_rom.v).

Self-checks: zero crossing, peak amplitude, symmetry (sum == 0 mod 2^16).
"""
import math

N = 256
AMPL = 32767
OUT = "src/rtl/sine_rom.v"

values = []
for i in range(N):
    v = round(AMPL * math.sin(2.0 * math.pi * i / N))
    values.append(v & 0xFFFF)

lines = [
    "// Auto-generated by tools/gen_sine.py - do not edit manually",
    "module sine_rom (",
    "    input  wire [7:0] addr,",
    "    output reg  [15:0] dout",
    ");",
    "    reg [15:0] mem [0:255];",
    "    initial begin",
]
for i in range(0, N, 4):
    row = "        " + " ".join(
        f"mem[{j:3d}] = 16'h{values[j]:04X};" for j in range(i, i + 4)
    )
    lines.append(row)
lines += [
    "    end",
    "    always @(*) dout = mem[addr];",
    "endmodule",
]
with open(OUT, "w") as f:
    f.write("\n".join(lines) + "\n")

# self-check
assert values[0] == 0 and values[128] == 0, "zero crossing wrong"
assert values[64] == 32767, "peak wrong"
assert values[192] == 32769, "negative peak wrong"   # (0-32767) & 0xFFFF
assert (sum(values) % 65536) == 0, "not zero-mean"
print(f"OK: {OUT} written, {N} entries, self-check passed")
```

`src/tb/dds_core_tb.v`(完整内容):
```verilog
`timescale 1ns / 1ps
module dds_core_tb;
    reg clk = 0;
    reg rst_n = 0;
    reg frame_edge = 0;
    reg [31:0] fword = 0;
    reg [1:0]  wave_sel = 0;
    wire [15:0] sample;

    dds_core uut (
        .clk(clk), .rst_n(rst_n), .frame_edge(frame_edge),
        .fword(fword), .wave_sel(wave_sel), .sample(sample)
    );
    always #40 clk = ~clk;   // 12.5 MHz

    integer errs = 0;
    integer i;

    function [15:0] sine_expected;
        input [7:0] a;
        real x;
        begin
            x = 32767.0 * $sin(2.0 * 3.14159265358979 * a / 256.0);
            if (x >= 0.0) sine_expected = $rtoi(x + 0.5);
            else          sine_expected = $rtoi(x - 0.5);
        end
    endfunction

    // drive one frame_edge pulse
    task frame;
        begin
            @(posedge clk); frame_edge = 1;
            @(posedge clk); frame_edge = 0;
            #10;
        end
    endtask

    initial begin
        #100 rst_n = 1;
        #100;

        // --- sine: fword = 2^24 -> phase advances one LUT step per frame ---
        wave_sel = 2'b00;
        fword = 32'h01000000;
        for (i = 0; i < 512; i = i + 1) begin
            frame;
            if ($signed(sample) !== $signed(sine_expected(i[7:0]))) begin
                $display("FAIL sine frame %0d: sample=%h exp=%h", i, sample, sine_expected(i[7:0]));
                errs = errs + 1;
            end
        end

        // --- square ---
        wave_sel = 2'b01;
        uut.phase_acc = 32'd0;     // poke internal state to reset phase
        uut.wave_sel_r = 2'b01;
        for (i = 0; i < 256; i = i + 1) begin
            frame;
            if (i < 128) begin
                if (sample !== 16'h7FFF) begin
                    $display("FAIL square frame %0d: %h", i, sample); errs = errs + 1;
                end
            end else begin
                if (sample !== 16'h8000) begin
                    $display("FAIL square frame %0d: %h", i, sample); errs = errs + 1;
                end
            end
        end

        // --- triangle ---
        wave_sel = 2'b10;
        uut.phase_acc = 32'd0;
        uut.wave_sel_r = 2'b10;
        for (i = 0; i < 256; i = i + 1) begin
            frame;
            begin
                reg [7:0] tri_p;
                reg [15:0] exp_v;
                // mirror DUT: tri_p = phase[31] ? ~phase[30:23] : phase[30:23]
                // with phase = i<<24: phase[30:23] = {i[6:0], 1'b0}
                tri_p = i[7] ? (~{i[6:0], 1'b0}) : {i[6:0], 1'b0};
                exp_v = (tri_p << 8) - 16'h8000;
                if (sample !== exp_v) begin
                    $display("FAIL tri frame %0d: %h exp %h", i, sample, exp_v);
                    errs = errs + 1;
                end
            end
        end

        // --- sawtooth ---
        wave_sel = 2'b11;
        uut.phase_acc = 32'd0;
        uut.wave_sel_r = 2'b11;
        for (i = 0; i < 256; i = i + 1) begin
            frame;
            begin
                reg [15:0] exp_v;
                exp_v = ({i[7:0], 8'h00}) - 16'h8000;
                if (sample !== exp_v) begin
                    $display("FAIL saw frame %0d: %h exp %h", i, sample, exp_v);
                    errs = errs + 1;
                end
            end
        end

        if (errs == 0) $display("PASS dds_core");
        else $display("FAIL dds_core: %0d errors", errs);
        $finish;
    end
endmodule
```

```verilog
`timescale 1ns / 1ps
module dds_core (
    input  wire clk,          // 12.288 MHz
    input  wire rst_n,
    input  wire frame_edge,   // one pulse per I2S frame
    input  wire [31:0] fword,
    input  wire [1:0]  wave_sel,   // 00 sin / 01 sqr / 10 tri / 11 saw
    output reg  [15:0] sample
);
    reg [31:0] phase_acc;
    reg [1:0]  wave_sel_r;

    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            phase_acc  <= 32'd0;
            wave_sel_r <= 2'b00;
        end else if (frame_edge) begin
            phase_acc  <= phase_acc + fword;
            wave_sel_r <= wave_sel;
        end

    wire [7:0] p = phase_acc[31:24];
    wire [15:0] sine_v;
    sine_rom u_rom (.addr(p), .dout(sine_v));
    wire [15:0] sqr_v = phase_acc[31] ? 16'h8000 : 16'h7FFF;   // phase 0..pi -> positive
    wire [7:0]  tri_p = phase_acc[31] ? ~phase_acc[30:23] : phase_acc[30:23];
    wire [15:0] tri_v = {tri_p, 8'h00} - 16'h8000;
    wire [15:0] saw_v = {p, 8'h00} - 16'h8000;

    always @(posedge clk or negedge rst_n)
        if (!rst_n) sample <= 16'd0;
        else if (frame_edge) begin
            case (wave_sel_r)
                2'b00:   sample <= sine_v;
                2'b01:   sample <= sqr_v;
                2'b10:   sample <= tri_v;
                default: sample <= saw_v;
            endcase
        end
endmodule
```

```verilog
`timescale 1ns / 1ps
module i2s_tx_tb;
    reg clk = 0;
    reg rst_n = 0;
    reg [15:0] l_data = 16'hAAAA;
    reg [15:0] r_data = 16'h5555;
    wire bclk, lrclk, dacdat, frame_edge;

    i2s_tx uut (
        .mclk(clk), .rst_n(rst_n),
        .l_data(l_data), .r_data(r_data),
        .bclk(bclk), .lrclk(lrclk), .dacdat(dacdat),
        .frame_edge(frame_edge)
    );
    always #40 clk = ~clk;   // 12.5 MHz

    integer errs = 0;
    reg [6:0] bcnt2 = 0;          // bclk rises since last frame_edge
    reg [15:0] gap = 0;
    integer frames = 0;
    reg [15:0] clk_cnt2 = 0;      // clk cycles since last lrclk edge
    reg [15:0] lr_period = 0;

    always @(posedge frame_edge) begin
        gap = bcnt2;
        bcnt2 = 0;
        frames = frames + 1;
    end
    always @(posedge bclk) bcnt2 = bcnt2 + 1;
    always @(posedge clk) clk_cnt2 = clk_cnt2 + 1;
    always @(posedge lrclk) begin
        lr_period = clk_cnt2;
        clk_cnt2 = 0;
    end

    // Verify one 16-bit word bit by bit: MSB is sampled at the 2nd bclk rising
    // edge after the lrclk edge (1-bclk gap), then bits 14..0 in order.
    task check_word;
        input [15:0] exp;
        input use_negedge;
        integer k;
        begin
            if (use_negedge) @(negedge lrclk);   // right word start (lrclk high->low)
            else             @(posedge lrclk);   // left word start (lrclk low->high)
            @(posedge bclk);                     // skip the gap (1st rise after edge)
            for (k = 15; k >= 0; k = k - 1) begin
                @(posedge bclk);
                if (dacdat !== exp[k]) begin
                    $display("FAIL i2s_tx bit %0d: got %b exp %b", k, dacdat, exp[k]);
                    errs = errs + 1;
                end
            end
        end
    endtask

    initial begin
        #100 rst_n = 1;
        #100_000;                      // ~9 LRCLK periods (each = 128 clk = 10.24 us)

        if (frames < 3) begin
            $display("FAIL i2s_tx: only %0d frames", frames); errs = errs + 1;
        end
        if (gap !== 32) begin
            $display("FAIL i2s_tx: %0d bclk per LRCLK period (exp 32)", gap); errs = errs + 1;
        end
        if (lr_period < 120 || lr_period > 136) begin
            $display("FAIL i2s_tx: lrclk period %0d clk (exp ~128)", lr_period); errs = errs + 1;
        end

        check_word(l_data, 1'b0);      // left word after lrclk rising edge
        check_word(r_data, 1'b1);      // right word after lrclk falling edge

        if (errs == 0) $display("PASS i2s_tx");
        else $display("FAIL i2s_tx: %0d errors", errs);
        $finish;
    end
endmodule
```

> 说明:数据字(16 bit)的采样点位于 LRCLK 边沿后第 2~17 个 BCLK 上升沿(第 1 个为间隙),与 ES8388 数据手册一致;LSB 落在下一半帧的第一个上升沿属标准行为。

`src/rtl/i2s_tx.v`(完整内容,1-BCLK 间隙版):
```verilog
`timescale 1ns / 1ps
// Philips I2S transmitter, 16-bit x 2ch, master clocks.
// mclk/4 -> bclk (3.072 MHz); 32 bclk per LRCLK period -> lrclk (96 kHz).
// Data changes on bclk falling edges; MSB is sampled at the 2nd bclk rising edge
// after the lrclk edge (1-bclk gap, per ES8388 I2S input timing).
module i2s_tx (
    input  wire mclk,          // 12.288 MHz
    input  wire rst_n,
    input  wire [15:0] l_data,
    input  wire [15:0] r_data,
    output reg  bclk,          // mclk/4 = 3.072 MHz
    output reg  lrclk,         // 96 kHz (toggles every 16 bclk)
    output reg  dacdat,
    output reg  frame_edge     // 1-clk pulse per LRCLK period (at L half start)
);
    reg [1:0] bcnt;            // mclk phase within bclk period
    reg [3:0] fcnt;            // bclk index within half-frame (0..15)
    reg [4:0] bitpos;          // data bit index within word (0 = MSB)
    reg [15:0] shift;
    reg lr;                    // current half-frame channel: 1 = left

    always @(posedge mclk or negedge rst_n)
        if (!rst_n) begin
            bcnt <= 2'd0;
            bclk <= 1'b0;
        end else begin
            bcnt <= bcnt + 2'd1;
            bclk <= bcnt[1];
        end

    always @(posedge mclk or negedge rst_n)
        if (!rst_n) begin
            fcnt <= 4'd0; lrclk <= 1'b1; bitpos <= 5'd0;
            shift <= 16'd0; lr <= 1'b1; frame_edge <= 1'b0;
        end else begin
            frame_edge <= 1'b0;
            if (bcnt == 2'd0) begin
                if (fcnt == 4'd15) begin
                    fcnt <= 4'd0;
                    lrclk <= ~lrclk;
                    bitpos <= 5'd0;
                    frame_edge <= (lr == 1'b0);   // once per LRCLK period (L half start)
                    shift <= (lr == 1'b1) ? r_data : l_data;
                    lr <= ~lr;
                end else begin
                    fcnt <= fcnt + 1'b1;
                    bitpos <= bitpos + 1'b1;
                end
            end
        end

    always @(posedge mclk or negedge rst_n)
        if (!rst_n) dacdat <= 1'b0;
        else if (bcnt == 2'd0) begin
            if (bitpos <= 5'd15) dacdat <= shift[15 - bitpos];
            else                 dacdat <= 1'b0;
        end
endmodule
```

时序说明(已按 ES8388 数据手册与第 22 章 Xilinx I2S IP 行为核实):LRCLK 与数据都在 BCLK 下降沿(bcnt==0)变化;边界处 LRCLK 翻转、装载新字,而 dacdat 保持旧字的 LSB(间隙值);MSB 在边界后下一个下降沿出现,被 LRCLK 边沿后的**第 2 个 BCLK 上升沿**采样,之后逐位到 LSB(第 17 个上升沿,即下一半帧的第一个上升沿)。

```verilog
`timescale 1ns / 1ps
module key_ctrl #(
    parameter DEBOUNCE_MS    = 20,
    parameter REPEAT_MS      = 500,
    parameter REPEAT_STEP_MS = 100
)(
    input  wire clk,          // 12.288 MHz
    input  wire rst_n,
    input  wire key_inc,      // S1, active low
    input  wire key_dec,      // S2, active low
    output reg  [15:0] freq_hz,
    output wire [31:0] fword,
    output wire [4:0]  freq_khz
);
    localparam [15:0] FREQ_MIN = 16'd20;
    localparam [15:0] FREQ_MAX = 16'd20000;
    localparam [4:0]  DB_MAX  = DEBOUNCE_MS[4:0] - 5'd1;
    localparam [9:0]  REPEAT_T  = REPEAT_MS[9:0];
    localparam [9:0]  REPEAT_STEP_T = REPEAT_STEP_MS[9:0];

    function [15:0] clamp;
        input [15:0] v;
        begin
            clamp = (v > FREQ_MAX) ? FREQ_MAX : ((v < FREQ_MIN) ? FREQ_MIN : v);
        end
    endfunction

    // 1 kHz tick: 12.288 MHz / 12288
    reg [13:0] tick_cnt;
    wire tick_1k = (tick_cnt == 14'd12287);
    always @(posedge clk or negedge rst_n)
        if (!rst_n) tick_cnt <= 14'd0;
        else        tick_cnt <= tick_1k ? 14'd0 : tick_cnt + 1'b1;

    // debounce: level flips after DEBOUNCE_MS stable ticks
    reg [4:0] db_cnt_inc, db_cnt_dec;
    reg key_inc_lvl, key_dec_lvl;
    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            db_cnt_inc <= 5'd0; db_cnt_dec <= 5'd0;
            key_inc_lvl <= 1'b1; key_dec_lvl <= 1'b1;
        end else if (tick_1k) begin
            if (key_inc == key_inc_lvl) db_cnt_inc <= 5'd0;
            else if (db_cnt_inc == DB_MAX) key_inc_lvl <= key_inc;
            else db_cnt_inc <= db_cnt_inc + 1'b1;
            if (key_dec == key_dec_lvl) db_cnt_dec <= 5'd0;
            else if (db_cnt_dec == DB_MAX) key_dec_lvl <= key_dec;
            else db_cnt_dec <= db_cnt_dec + 1'b1;
        end

    // press edge + hold/repeat
    reg key_inc_p, key_dec_p;
    reg [1:0] active;        // 0 none / 1 inc / 2 dec
    reg [9:0] hold_ms;
    reg rep_mode;
    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            freq_hz <= 16'd1000;
            key_inc_p <= 1'b1; key_dec_p <= 1'b1;
            active <= 2'd0; hold_ms <= 10'd0; rep_mode <= 1'b0;
        end else if (tick_1k) begin
            key_inc_p <= key_inc_lvl;
            key_dec_p <= key_dec_lvl;
            if (!key_inc_lvl && key_inc_p) begin
                active <= 2'd1; hold_ms <= 10'd0; rep_mode <= 1'b0;
                freq_hz <= clamp(freq_hz + 16'd10);
            end else if (!key_dec_lvl && key_dec_p) begin
                active <= 2'd2; hold_ms <= 10'd0; rep_mode <= 1'b0;
                freq_hz <= (freq_hz > 16'd10) ? clamp(freq_hz - 16'd10) : FREQ_MIN;
            end
            if (key_inc_lvl && key_dec_lvl) begin
                active <= 2'd0; hold_ms <= 10'd0; rep_mode <= 1'b0;
            end
            if (active == 2'd1 && !key_inc_lvl) begin
                if (!rep_mode && hold_ms == REPEAT_T) begin
                    rep_mode <= 1'b1; hold_ms <= 10'd0;
                    freq_hz <= clamp(freq_hz + 16'd200);
                end else if (rep_mode && hold_ms == REPEAT_STEP_T) begin
                    hold_ms <= 10'd0;
                    freq_hz <= clamp(freq_hz + 16'd200);
                end else
                    hold_ms <= hold_ms + 1'b1;
            end else if (active == 2'd2 && !key_dec_lvl) begin
                if (!rep_mode && hold_ms == REPEAT_T) begin
                    rep_mode <= 1'b1; hold_ms <= 10'd0;
                    freq_hz <= (freq_hz > 16'd200) ? clamp(freq_hz - 16'd200) : FREQ_MIN;
                end else if (rep_mode && hold_ms == REPEAT_STEP_T) begin
                    hold_ms <= 10'd0;
                    freq_hz <= (freq_hz > 16'd200) ? clamp(freq_hz - 16'd200) : FREQ_MIN;
                end else
                    hold_ms <= hold_ms + 1'b1;
            end
        end

    // fword = round(freq * 2^32 / 96000), exact via 44739 + 23296/96000
    assign fword = freq_hz * 32'd44739 + ((freq_hz * 32'd23296 + 32'd48000) / 32'd96000);
    assign freq_khz = freq_hz / 16'd1000;
endmodule
```

```verilog
// Expects byte stream per write: [0x20][reg][data] (+ACK after each).
module i2c_slave_mon (
    input  wire clk,
    input  wire rst_n,
    inout  wire scl,
    inout  wire sda,
    output reg [7:0] mon_reg,
    output reg [7:0] mon_data,
    output reg wr_valid      // 1-clk pulse when a full write (reg+data) captured
);
    reg scl_s1, scl_s2, sda_s1, sda_s2;
    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin scl_s1<=1'b1; scl_s2<=1'b1; sda_s1<=1'b1; sda_s2<=1'b1; end
        else begin
            scl_s1 <= scl; scl_s2 <= scl_s1;
            sda_s1 <= sda; sda_s2 <= sda_s1;
        end

    wire scl_rise =  scl_s2 == 1'b0 && scl_s1 == 1'b1;
    wire scl_fall =  scl_s2 == 1'b1 && scl_s1 == 1'b0;
    wire sda_fall =  sda_s2 == 1'b1 && sda_s1 == 1'b0;
    wire sda_rise =  sda_s2 == 1'b0 && sda_s1 == 1'b1;
    wire start = scl_s1 && sda_fall;
    wire stop  = scl_s1 && sda_rise;

    reg [3:0] bitcnt;
    reg [7:0] shift;
    reg [1:0] bytecnt;
    reg sda_o, oe;
    reg ack_asserted;
    assign sda = oe ? sda_o : 1'bz;

    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            bitcnt <= 4'd0; shift <= 8'd0; bytecnt <= 2'd0;
            mon_reg <= 8'd0; mon_data <= 8'd0; wr_valid <= 1'b0;
            sda_o <= 1'b1; oe <= 1'b0; ack_asserted <= 1'b0;
        end else begin
            wr_valid <= 1'b0;
            if (start || stop) begin
                bitcnt <= 4'd0; bytecnt <= 2'd0; oe <= 1'b0; sda_o <= 1'b1;
                ack_asserted <= 1'b0;
            end else if (scl_rise) begin
                if (bitcnt < 4'd8) begin

                            default: ;
                        endcase
                        if (bytecnt != 2'd2) bytecnt <= bytecnt + 1'b1;
                    end
                    bitcnt <= bitcnt + 1'b1;
                end
            end else if (scl_fall) begin
                if (bitcnt == 4'd8) begin
                    if (!ack_asserted) begin
                        oe <= 1'b1; sda_o <= 1'b0; ack_asserted <= 1'b1;   // ACK during 9th bit
                    end else begin
                        oe <= 1'b0; sda_o <= 1'b1; bitcnt <= 4'd0;          // release after 9th bit
                        ack_asserted <= 1'b0;
                    end
                end
            end
        end
```

```verilog
// Power-on ES8388 configuration: replays the 22nd-chapter register table
// (39 writes) through the ch20 I2C controller, then raises cfg_done.
module es8388_cfg (
    input  wire clk,          // 12.288 MHz
    input  wire rst_n,
    output reg  cfg_done,
    output reg  cfg_err,      // any NACK during config
    output reg  wrreg_req,
    output wire [15:0] reg_addr,   // {8'h00, reg}
    output wire [7:0]  reg_data,
    input  wire rw_done,
    input  wire ack_err
);
    localparam [5:0] NREG = 6'd39;
    localparam [19:0] WAIT_20MS = 20'd245_760;   // 12.288M * 0.02s

    function [15:0] rom_get;
        input [5:0] i;
        begin
            case (i)
                6'd0 : rom_get = 16'h0080;   // 0x00 <- 0x80 soft reset
                6'd1 : rom_get = 16'h0000;   // 0x00 <- 0x00
                6'd2 : rom_get = 16'h0158;
                6'd3 : rom_get = 16'h0150;
                6'd4 : rom_get = 16'h02F3;
                6'd5 : rom_get = 16'h02F0;
                6'd6 : rom_get = 16'h2B80;   // ADC/DAC share LRCLK
                6'd7 : rom_get = 16'h0036;
                6'd8 : rom_get = 16'h0800;   // I2S slave mode
                6'd9 : rom_get = 16'h0308;
                6'd10: rom_get = 16'h0400;
                6'd11: rom_get = 16'h0D00;   // MCLK/fs = 128 -> 96 kHz
                6'd12: rom_get = 16'h1800;
                6'd13: rom_get = 16'h0500;
                6'd14: rom_get = 16'h06C3;
                6'd15: rom_get = 16'h0A00;
                6'd16: rom_get = 16'h0B02;
                6'd17: rom_get = 16'h0C0C;   // ADC: I2S, 16-bit
                6'd18: rom_get = 16'h1718;   // DAC: I2S, 16-bit
                6'd19: rom_get = 16'h1000;
                6'd20: rom_get = 16'h1100;
                6'd21: rom_get = 16'h1A00;
                6'd22: rom_get = 16'h1B00;
                6'd23: rom_get = 16'h0900;
                6'd24: rom_get = 16'h1211;   // disable ALC
                6'd25: rom_get = 16'h13C0;
                6'd26: rom_get = 16'h1432;
                6'd27: rom_get = 16'h1506;
                6'd28: rom_get = 16'h16C3;
                6'd29: rom_get = 16'h27B8;
                6'd30: rom_get = 16'h2AB8;
                6'd31: rom_get = 16'h0200;   // pause 20 ms after this
                6'd32: rom_get = 16'h2E1E;
                6'd33: rom_get = 16'h2F1E;
                6'd34: rom_get = 16'h301E;
                6'd35: rom_get = 16'h311E;
                6'd36: rom_get = 16'h0436;   // DAC power + LOUT1/ROUT1 + OUT2
                6'd37: rom_get = 16'h2600;
                6'd38: rom_get = 16'h0309;
                default: rom_get = 16'h0000;
            endcase
        end
    endfunction

    localparam [1:0] S_IDLE = 2'd0, S_WRITE = 2'd1, S_WAIT = 2'd2, S_PAUSE = 2'd3;
    reg [1:0] state;
    reg [5:0] idx;
    reg [19:0] wait_cnt;

    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            cfg_done <= 1'b0; cfg_err <= 1'b0;
            wrreg_req <= 1'b0;
            reg_addr <= 16'd0; reg_data <= 8'd0;
            idx <= 6'd0; wait_cnt <= 20'd0;

                        reg_data <= rom_get(idx)[7:0];

                    state <= S_WAIT;

                S_WAIT: begin

                        if (ack_err) cfg_err <= 1'b1;
                        if (idx == NREG - 6'd1) begin
                            cfg_done <= 1'b1;
                            state <= S_IDLE;
                        end else begin
                            idx <= idx + 1'b1;
                            if (idx == 6'd31) begin
                                wait_cnt <= 20'd0;
                                state <= S_PAUSE;
                            end else
                                state <= S_IDLE;
                        end
                    end
                end
                S_PAUSE: begin
                    if (wait_cnt == WAIT_20MS) begin
                        wait_cnt <= 20'd0;
                        state <= S_IDLE;
                    end else
                        wait_cnt <= wait_cnt + 1'b1;
                end
                default: state <= S_IDLE;
            endcase
        end
endmodule
```

```verilog
        .clk(clk_12m288), .rst_n(rst_in_n), .scl(i2c_scl), .sda(i2c_sda),
        .mon_reg(mon_reg), .mon_data(mon_data), .wr_valid(wr_valid)
    );
    always #40 clk_12m288 = ~clk_12m288;   // 12.5 MHz

    integer errs = 0;
    integer n = 0;
    integer frames = 0;
    reg [31:0] l_cap = 0;
    integer frame_gap = -1;
    reg [6:0] bcnt2 = 0;
    integer i;

    always @(posedge wr_valid) n = n + 1;
    always @(posedge i2s_bclk) bcnt2 = bcnt2 + 1;
    always @(posedge uut.frame_edge) begin
        frame_gap = bcnt2; bcnt2 = 0; frames = frames + 1;
    end
    always @(posedge i2s_bclk) begin
        if (i2s_lrclk) l_cap <= {l_cap[30:0], i2s_dacdat};
    end

    initial begin
        #100 rst_in_n = 1;
        // wait for ES8388 config to complete (~31 ms simulated)
        wait (uut.cfg_done == 1'b1);
        #2000;

        if (n !== 39) begin
            $display("FAIL awg_top: %0d I2C writes, exp 39", n); errs = errs + 1;
        end
        if (uut.cfg_err !== 1'b0) begin
            $display("FAIL awg_top: cfg_err set"); errs = errs + 1;
        end
        // default freq 1 kHz -> freq_khz = 1 -> led[7:2] = {0, 00001}
        if (led !== 8'b0000_0010) begin
            $display("FAIL awg_top: led=%b (exp 00000010 at 1kHz, sw=00)", led); errs = errs + 1;
        end
        // frames running at ~96 kHz: measure over 10 ms
        frames = 0;
        #10_000_000;
        if (frames < 900 || frames > 1020) begin
            $display("FAIL awg_top: %0d frames in 10ms (exp ~960)", frames); errs = errs + 1;
        end
        // dacdat active (sample transitions): l_cap must not be all ones/zeros
        if (l_cap === 32'h00000000 || l_cap === 32'hFFFFFFFF) begin
            $display("FAIL awg_top: no data activity on I2S"); errs = errs + 1;
        end
        // short press S1: 1000 -> 1010 Hz
        key_s1 = 0; #8_000_000; key_s1 = 1; #2_000_000;
        if (uut.freq_hz !== 16'd1010) begin
            $display("FAIL awg_top: freq=%0d exp 1010", uut.freq_hz); errs = errs + 1;
        end
        if (led[6:2] !== 5'b00001) begin
            $display("FAIL awg_top: led after key press=%b", led); errs = errs + 1;
        end
        // waveform switch -> LED follows
        sw = 2'b10;
        #100_000;
        if (led[1:0] !== 2'b10) begin
            $display("FAIL awg_top: led wave=%b", led[1:0]); errs = errs + 1;
        end

        if (errs == 0) $display("PASS awg_top");
        else $display("FAIL awg_top: %0d errors", errs);
        $finish;
    end
endmodule
```
