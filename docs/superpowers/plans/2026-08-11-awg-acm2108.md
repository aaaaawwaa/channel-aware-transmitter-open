```python
#!/usr/bin/env python3
"""Generate a 256 x 8-bit signed sine ROM as Verilog (src/rtl/sine_rom.v).

Self-checks: zero crossing, peak amplitude, symmetry (sum == 0 mod 2^8).
"""
import math

N = 256
AMPL = 127
OUT = "src/rtl/sine_rom.v"

values = []
for i in range(N):
    v = round(AMPL * math.sin(2.0 * math.pi * i / N))
    values.append(v & 0xFF)

lines = [
    "// Auto-generated by tools/gen_sine.py - do not edit manually",
    "module sine_rom (",
    "    input  wire [7:0] addr,",
    "    output reg  [7:0] dout",
    ");",
    "    reg [7:0] mem [0:255];",
    "    initial begin",
]
for i in range(0, N, 4):
    row = "        " + " ".join(
        f"mem[{j:3d}] = 8'h{values[j]:02X};" for j in range(i, i + 4)
    )
    lines.append(row)
lines += [
    "    end",
    "    always @(*) dout = mem[addr];",
    "endmodule",
]
with open(OUT, "w") as f:
    f.write("\n".join(lines) + "\n")

# self-check
assert values[0] == 0 and values[128] == 0, "zero crossing wrong"
assert values[64] == 127, "peak wrong"
assert values[192] == 129, "negative peak wrong"   # (0-127) & 0xFF
assert (sum(values) % 256) == 0, "not zero-mean"
print(f"OK: {OUT} written, {N} entries, self-check passed")
```

```verilog
`timescale 1ns / 1ps
module dds_core_tb;
    reg clk = 0;
    reg rst_n = 0;
    reg [31:0] fword = 0;
    reg [1:0]  wave_sel = 0;
    wire [7:0] sample;

    dds_core uut (
        .clk(clk), .rst_n(rst_n),
        .fword(fword), .wave_sel(wave_sel), .sample(sample)
    );
    always #10 clk = ~clk;   // 50 MHz

    integer errs = 0;
    integer i;

    function [7:0] sine_expected;
        input [7:0] a;
        real x;
        begin
            x = 127.0 * $sin(2.0 * 3.14159265358979 * a / 256.0);
            if (x >= 0.0) sine_expected = $rtoi(x + 0.5);
            else          sine_expected = $rtoi(x - 0.5);
        end
    endfunction

    // wait one clk edge, then check sample (registered at posedge)
    task step;
        begin
            @(posedge clk);
            #1;
        end
    endtask

    initial begin
        #100 rst_n = 1;
        #100;

        // --- sine: fword = 2^24 -> one LUT step per clk ---
        wave_sel = 2'b00;
        fword = 32'h01000000;
        for (i = 0; i < 512; i = i + 1) begin
            step;
            if ($signed(sample) !== $signed(sine_expected(i[7:0]))) begin
                $display("FAIL sine step %0d: sample=%h exp=%h", i, sample, sine_expected(i[7:0]));
                errs = errs + 1;
            end
        end

        // --- square ---
        wave_sel = 2'b01;
        uut.phase_acc = 32'd0;
        uut.wave_sel_r = 2'b01;
        for (i = 0; i < 256; i = i + 1) begin
            step;
            if (i < 128) begin
                if (sample !== 8'h7F) begin
                    $display("FAIL square step %0d: %h", i, sample); errs = errs + 1;
                end
            end else begin
                if (sample !== 8'h81) begin
                    $display("FAIL square step %0d: %h", i, sample); errs = errs + 1;
                end
            end
        end

        // --- triangle ---
        wave_sel = 2'b10;
        uut.phase_acc = 32'd0;
        uut.wave_sel_r = 2'b10;
        for (i = 0; i < 256; i = i + 1) begin
            step;
            begin
                reg [7:0] tri_p;
                reg [7:0] exp_v;
                // mirror DUT: tri_p = phase[31] ? ~phase[30:23] : phase[30:23]
                // with phase = i<<24: phase[30:23] = {i[6:0], 1'b0}
                tri_p = i[7] ? (~{i[6:0], 1'b0}) : {i[6:0], 1'b0};
                exp_v = tri_p - 8'h80;
                if (sample !== exp_v) begin
                    $display("FAIL tri step %0d: %h exp %h", i, sample, exp_v);
                    errs = errs + 1;
                end
            end
        end

        // --- sawtooth ---
        wave_sel = 2'b11;
        uut.phase_acc = 32'd0;
        uut.wave_sel_r = 2'b11;
        for (i = 0; i < 256; i = i + 1) begin
            step;
            begin
                reg [7:0] exp_v;
                exp_v = i[7:0] - 8'h80;
                if (sample !== exp_v) begin
                    $display("FAIL saw step %0d: %h exp %h", i, sample, exp_v);
                    errs = errs + 1;
                end
            end
        end

        if (errs == 0) $display("PASS dds_core");
        else $display("FAIL dds_core: %0d errors", errs);
        $finish;
    end
endmodule
```

`timescale 1ns / 1ps
module dds_core (
    input  wire clk,          // 50 MHz
    input  wire rst_n,
    input  wire [31:0] fword,
    input  wire [1:0]  wave_sel,   // 00 sin / 01 sqr / 10 tri / 11 saw
    output reg  [7:0] sample
);
    reg [31:0] phase_acc;
    reg [1:0]  wave_sel_r;

    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            phase_acc  <= 32'd0;
            wave_sel_r <= 2'b00;
        end else begin
            phase_acc  <= phase_acc + fword;
            wave_sel_r <= wave_sel;
        end

    wire [7:0] p = phase_acc[31:24];
    wire [7:0] sine_v;
    sine_rom u_rom (.addr(p), .dout(sine_v));
    wire [7:0] sqr_v = phase_acc[31] ? 8'h81 : 8'h7F;   // -127 / +127
    wire [7:0] tri_p = phase_acc[31] ? ~phase_acc[30:23] : phase_acc[30:23];
    wire [7:0] tri_v = tri_p - 8'h80;
    wire [7:0] saw_v = p - 8'h80;

    always @(posedge clk or negedge rst_n)
        if (!rst_n) sample <= 8'd0;
        else begin
            case (wave_sel_r)
                2'b00:   sample <= sine_v;
                2'b01:   sample <= sqr_v;
                2'b10:   sample <= tri_v;
                default: sample <= saw_v;
            endcase
        end
endmodule
```

module key_ctrl_tb;
    reg clk = 0;
    reg rst_n = 0;
    reg key_inc = 1, key_dec = 1;
    wire [19:0] freq_hz;
    wire [31:0] fword;
    wire [9:0]  freq_khz;

    key_ctrl #(.DEBOUNCE_MS(3), .REPEAT_MS(50), .REPEAT_STEP_MS(10)) uut (
        .clk(clk), .rst_n(rst_n), .key_inc(key_inc), .key_dec(key_dec),
        .freq_hz(freq_hz), .fword(fword), .freq_khz(freq_khz)
    );
    always #10 clk = ~clk;   // 50 MHz (tick = 50000 clk = 1 ms, exact)

    integer errs = 0;
    integer step = 0;

    function [31:0] fword_exp;
        input [19:0] f;
        begin
            fword_exp = f * 20'd85 + ((f * 20'd9) / 20'd10);
        end
    endfunction

    task chk;
        input [19:0] exp_freq;
        begin
            step = step + 1;
            if (freq_hz !== exp_freq) begin
                $display("FAIL key_ctrl step %0d: freq=%0d exp=%0d", step, freq_hz, exp_freq);
                errs = errs + 1;
            end
            if (fword !== fword_exp(exp_freq)) begin
                $display("FAIL key_ctrl step %0d: fword=%0d exp=%0d", step, fword, fword_exp(exp_freq));
                errs = errs + 1;
            end
            if (freq_khz !== exp_freq / 1000) begin
                $display("FAIL key_ctrl step %0d: freq_khz=%0d exp=%0d", step, freq_khz, exp_freq / 1000);
                errs = errs + 1;
            end
        end
    endtask

    task press_inc; input integer ms; begin key_inc = 0; #(ms * 1_000_000); key_inc = 1; #5_000_000; end endtask
    task press_dec; input integer ms; begin key_dec = 0; #(ms * 1_000_000); key_dec = 1; #5_000_000; end endtask

    initial begin
        #100 rst_n = 1;
        #5_000_000;
        chk(20'd1000);                    // reset default 1 kHz

        press_inc(5);                     // short press -> +10
        chk(20'd1010);

        press_inc(70);                    // long: +10 @press, +200 @50ms, +200 @60ms
        chk(20'd1420);

        press_inc(1);                     // glitch (< debounce) -> no change
        chk(20'd1420);

        press_dec(200);                   // -10 then repeated -200 -> clamp at 10
        chk(20'd10);

        press_dec(30);                    // already at min
        chk(20'd10);

        // --- upper clamp: poke near max, then long-press inc ---
        uut.freq_hz = 20'd999_990;
        #1_000_000;
        chk(20'd999_990);
        press_inc(60);                    // +10 -> 1,000,000, then repeats clamp at max
        chk(20'd1_000_000);

        if (errs == 0) $display("PASS key_ctrl");
        else $display("FAIL key_ctrl: %0d errors", errs);
        $finish;
    end

`timescale 1ns / 1ps
module key_ctrl #(
    parameter DEBOUNCE_MS    = 20,
    parameter REPEAT_MS      = 500,
    parameter REPEAT_STEP_MS = 100
)(
    input  wire clk,          // 50 MHz
    input  wire rst_n,
    input  wire key_inc,      // S1, active low
    input  wire key_dec,      // S2, active low
    output reg  [19:0] freq_hz,
    output wire [31:0] fword,
    output wire [9:0]  freq_khz
);
    localparam [19:0] FREQ_MIN = 20'd10;
    localparam [19:0] FREQ_MAX = 20'd1_000_000;
    localparam [4:0]  DB_MAX  = DEBOUNCE_MS[4:0] - 5'd1;
    localparam [9:0]  REPEAT_T  = REPEAT_MS[9:0];
    localparam [9:0]  REPEAT_STEP_T = REPEAT_STEP_MS[9:0];

    function [19:0] clamp;
        input [19:0] v;
        begin
            clamp = (v > FREQ_MAX) ? FREQ_MAX : ((v < FREQ_MIN) ? FREQ_MIN : v);
        end
    endfunction

    // 1 kHz tick: 50 MHz / 50000 = 1 ms exactly
    reg [15:0] tick_cnt;
    wire tick_1k = (tick_cnt == 16'd49999);
    always @(posedge clk or negedge rst_n)
        if (!rst_n) tick_cnt <= 16'd0;
        else        tick_cnt <= tick_1k ? 16'd0 : tick_cnt + 1'b1;

    // debounce: level flips after DEBOUNCE_MS stable ticks
    reg [4:0] db_cnt_inc, db_cnt_dec;
    reg key_inc_lvl, key_dec_lvl;
    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            db_cnt_inc <= 5'd0; db_cnt_dec <= 5'd0;
            key_inc_lvl <= 1'b1; key_dec_lvl <= 1'b1;
        end else if (tick_1k) begin
            if (key_inc == key_inc_lvl) db_cnt_inc <= 5'd0;
            else if (db_cnt_inc == DB_MAX) key_inc_lvl <= key_inc;
            else db_cnt_inc <= db_cnt_inc + 1'b1;
            if (key_dec == key_dec_lvl) db_cnt_dec <= 5'd0;
            else if (db_cnt_dec == DB_MAX) key_dec_lvl <= key_dec;
            else db_cnt_dec <= db_cnt_dec + 1'b1;
        end

    // press edge + hold/repeat
    reg key_inc_p, key_dec_p;
    reg [1:0] active;        // 0 none / 1 inc / 2 dec
    reg [9:0] hold_ms;
    reg rep_mode;
    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            freq_hz <= 20'd1000;
            key_inc_p <= 1'b1; key_dec_p <= 1'b1;
            active <= 2'd0; hold_ms <= 10'd0; rep_mode <= 1'b0;
        end else if (tick_1k) begin
            key_inc_p <= key_inc_lvl;
            key_dec_p <= key_dec_lvl;
            if (!key_inc_lvl && key_inc_p) begin
                active <= 2'd1; hold_ms <= 10'd0; rep_mode <= 1'b0;
                freq_hz <= clamp(freq_hz + 20'd10);
            end else if (!key_dec_lvl && key_dec_p) begin
                active <= 2'd2; hold_ms <= 10'd0; rep_mode <= 1'b0;
                freq_hz <= (freq_hz > 20'd10) ? clamp(freq_hz - 20'd10) : FREQ_MIN;
            end
            if (key_inc_lvl && key_dec_lvl) begin
                active <= 2'd0; hold_ms <= 10'd0; rep_mode <= 1'b0;
            end
            if (active == 2'd1 && !key_inc_lvl) begin
                if (!rep_mode && hold_ms == REPEAT_T) begin
                    rep_mode <= 1'b1; hold_ms <= 10'd0;
                    freq_hz <= clamp(freq_hz + 20'd200);
                end else if (rep_mode && hold_ms == REPEAT_STEP_T) begin
                    hold_ms <= 10'd0;
                    freq_hz <= clamp(freq_hz + 20'd200);
                end else
                    hold_ms <= hold_ms + 1'b1;
            end else if (active == 2'd2 && !key_dec_lvl) begin
                if (!rep_mode && hold_ms == REPEAT_T) begin
                    rep_mode <= 1'b1; hold_ms <= 10'd0;
                    freq_hz <= (freq_hz > 20'd200) ? clamp(freq_hz - 20'd200) : FREQ_MIN;
                end else if (rep_mode && hold_ms == REPEAT_STEP_T) begin
                    hold_ms <= 10'd0;
                    freq_hz <= (freq_hz > 20'd200) ? clamp(freq_hz - 20'd200) : FREQ_MIN;
                end else
                    hold_ms <= hold_ms + 1'b1;
            end
        end

    // fword = freq*85 + (freq*9)/10  ~  freq * 2^32 / 50e6 (error < 0.001%)
    assign fword = freq_hz * 20'd85 + ((freq_hz * 20'd9) / 20'd10);
    assign freq_khz = freq_hz / 20'd1000;
endmodule
```

`timescale 1ns / 1ps
module dac_driver_tb;
    reg clk = 0;
    reg rst_n = 0;
    reg [7:0] sample = 8'h00;
    wire [7:0] dac0_db, dac1_db;
    wire daclk0, daclk1;

    dac_driver uut (
        .clk(clk), .rst_n(rst_n), .sample(sample),
        .dac0_db(dac0_db), .dac1_db(dac1_db),
        .daclk0(daclk0), .daclk1(daclk1)
    );
    always #10 clk = ~clk;   // 50 MHz

    integer errs = 0;
    integer clk_rises = 0;
    integer daclk_rises = 0;
    integer window_rises = 0;
    reg [7:0] exp_d = 8'h00;
    reg window_en = 0;

    always @(posedge clk) begin
        exp_d <= sample;                 // data expected on DAC pins 1 clk later
        clk_rises = clk_rises + 1;
    end
    always @(posedge daclk0) begin
        daclk_rises = daclk_rises + 1;
        if (window_en) window_rises = window_rises + 1;
        if (rst_n) begin                 // guard t=0 X-state edge
            if (dac0_db !== exp_d) begin
                $display("FAIL dac_driver: data at daclk rise %h exp %h", dac0_db, exp_d);
                errs = errs + 1;
            end
            if (dac0_db !== dac1_db) begin
                $display("FAIL dac_driver: ch0 %h != ch1 %h", dac0_db, dac1_db);
                errs = errs + 1;
            end
            if (daclk0 !== daclk1) begin
                $display("FAIL dac_driver: clk0 != clk1"); errs = errs + 1;
            end
        end
    end

    initial begin
        #100 rst_n = 1;
        #100;
        // ramp the sample
        repeat (100) begin
            @(posedge clk); #1;
            sample = sample + 8'd3;
        end
        #1000;
        // daclk must run at 50 MHz: one rise per clk period -> ~1000 in the window
        window_en = 1;
        repeat (1000) @(posedge clk);
        window_en = 0;
        if (window_rises < 950 || window_rises > 1050) begin
            $display("FAIL dac_driver: %0d daclk rises in 1000 clk", window_rises);
            errs = errs + 1;
        end
        if (errs == 0) $display("PASS dac_driver");
        else $display("FAIL dac_driver: %0d errors", errs);
        $finish;
    end
endmodule
```

`timescale 1ns / 1ps
// Parallel DAC driver for MS9708 (latch on rising edge).
// Data registered on posedge clk; DACLK = ~clk (combinational) -> its rising
// edge occurs at the clk falling edge, 10 ns after the data update (half a
// 50 MHz period), ample setup time. DAC0 and DAC1 receive identical data and
// clock (dual-channel sync output).
module dac_driver (
    input  wire clk,          // 50 MHz
    input  wire rst_n,
    input  wire [7:0] sample,
    output reg  [7:0] dac0_db,
    output reg  [7:0] dac1_db,
    output wire daclk0,
    output wire daclk1
);
    always @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            dac0_db <= 8'd0;
            dac1_db <= 8'd0;
        end else begin
            dac0_db <= sample;
            dac1_db <= sample;
        end

    // NOTE: cannot register ~clk on posedge clk (would latch constant 0);
    // combinational inversion gives the rising edge at the clk falling edge.
    assign daclk0 = ~clk;
    assign daclk1 = ~clk;
endmodule
```
